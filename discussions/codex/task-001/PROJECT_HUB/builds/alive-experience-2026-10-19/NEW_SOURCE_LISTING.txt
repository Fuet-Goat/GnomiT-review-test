		return
	if reduced_motion:
		# Events still show a real pose; idle has no periodic blinking or breathing.
		_blend = 1.0
		if _state == "blink":
			_state = "idle"
			_set_pose(0, 0.0)
		elif _state == "reaction":
			_state_time -= delta
			if _state_time <= 0.0:
				_state = "idle"
				_set_pose(0, 0.0)
		queue_redraw()
		return
	_clock += delta
	if _blend < 1.0:
		_blend = minf(1.0, _blend + delta / _blend_duration)
	if _state == "victory":
		pass
	elif _state != "idle":

		if _state_time <= 0.0:
			_state = "idle"

			_blink_index += 1
