+func setup_lab(scenario: String, seed_value: int = 42) -> void:
+	black_hole_used = false
+	setup({
+		"cols": 7, "rows": 8, "moves": 12,
+		"goal": "collect", "collect_kind": 3, "collect_goal": 999,
+		"mask": ["xxxxxxx", "xxxxxxx", "xxxxxxx", "xxx.xxx", "xx.xxxx", "xxxxxxx", "xxxxxxx", "xxxxxxx"],

+		"tornado_v":

+func try_swap(a: int, b: int) -> Dictionary:
+	if won or lost or not _can_swap(a, b):
+		return _result(false, [])
+	var kind_a := grid[a]
+	var kind_b := grid[b]
+	if not _is_special(kind_a) and not _is_special(kind_b):
+		# Ordinary swaps retain the game's existing collection, soil and rune rules.
+		return super.try_swap(a, b)
+	var combination := _is_black_hole_pair(kind_a, kind_b)
+	if combination and black_hole_used:
+		return _result(false, [])
+	if not combination and _is_special(kind_a) and _is_special(kind_b):
+		return _result(false, [])
+	var events: Array = []
+	var before := grid.duplicate()
+	_exchange(a, b)
+	events.append(_event("swap", before, {"a": a, "b": b}))
+	moves_left -= 1
+	var origin := b if _is_special(kind_a) else a
+	var power := "black_hole" if combination else _power_name(grid[origin])
+	var targets: Array[int] = []
+	if combination:
+		black_hole_used = true
+		for index in range(grid.size()):
+			if _is_ordinary(grid[index]):
+				targets.append(index)
+	elif power == "volcano":
+		var candidates: Array[int] = []
+		for index in range(grid.size()):
+			if _is_ordinary(grid[index]):
+				candidates.append(index)
+		var hit_count := mini(_rng.randi_range(3, 5), candidates.size())
+		for hit in range(hit_count):
+			var selected := _rng.randi_range(0, candidates.size() - 1)
+			targets.append(candidates[selected])
+			candidates.remove_at(selected)
+		targets.sort()
+	else:
+		for index in range(grid.size()):
+			var aligned := index / cols == origin / cols if power == "tornado_h" else index % cols == origin % cols
+			if aligned and _is_ordinary(grid[index]):
+				targets.append(index)
+	_apply_power(power, origin, targets, events)
+	_resolve_after_power(events)
+	return _result(true, events)

+func _apply_power(power: String, origin: int, targets: Array[int], events: Array) -> void:
+	var before := grid.duplicate()
+	var floor_before := floor.duplicate()
+	var runes_before := runes.duplicate()
+	var removed: Array[int] = []
+	if power == "black_hole":
+		# Other specials are consumed inertly, never recursively activated.
+		for index in range(grid.size()):
+			if grid[index] >= 0:
+				removed.append(index)
+	else:
+		removed.assign(targets)
+		removed.append(origin)
+		removed.sort()
+	for index in removed:
+		if _is_ordinary(grid[index]):
+			if grid[index] == collect_kind:
+				collected += 1
+			if power != "black_hole":
+				floor[index] = maxi(0, floor[index] - 1)
+		grid[index] = -1
+	# Direct effects never decrement neighboring runes. Normal clears after refill can.
+	events.append(_event("natural_power", before, {
+		"power": power, "origin": origin,
+		"targets": targets.duplicate(), "indices": removed,
+		"floor_before": floor_before, "runes_before": runes_before,
+	}))

+func _resolve_after_power(events: Array) -> void:
+	_fall(events)
+	var matches := get_matches()
+	var wave := 0
+	while not matches.is_empty() and wave < MAX_CASCADE_WAVES:
+		_clear(matches, events)
+		_fall(events)
+		matches = get_matches()
+		wave += 1
+	if not matches.is_empty() or _find_move().is_empty():
+		var before := grid.duplicate()
+		var playable := _make_playable()

+		if not playable:
+			lost = true
+	# Resolve the real goal after every cascade, including on the final move.
+	won = _goal_complete()
+	lost = not won and (lost or moves_left <= 0)

+func get_matches() -> Array[int]:
+	var matches := super.get_matches()
+	var ordinary: Array[int] = []
+	for index in matches:
+		if _is_ordinary(grid[index]):
+			ordinary.append(index)
+	return ordinary

+func _has_match_at(index: int) -> bool:
+	return index >= 0 and index < grid.size() and _is_ordinary(grid[index]) and super._has_match_at(index)
+
+func _find_move() -> Array:
+	for index in range(grid.size()):
+		if grid[index] < 0:
+			continue
+		for neighbor in [index + 1, index + cols]:
+			if not _can_swap(index, neighbor):
+				continue
+			var first := grid[index]
+			var second := grid[neighbor]
+			if _is_black_hole_pair(first, second):
+				if not black_hole_used:
+					return [index, neighbor]
+				continue
+			if _is_special(first) != _is_special(second):
+				return [index, neighbor]
+			if _is_special(first) or _is_special(second) or first == second:
+				continue
+			_exchange(index, neighbor)
+			var matches := _has_match_at(index) or _has_match_at(neighbor)
+			_exchange(index, neighbor)
+			if matches:
+				return [index, neighbor]
+	return []
+
+func _fill_stable() -> void:
+	# The base reshuffle may rewrite all nonnegative cells. Keep surviving objects.
+	for index in range(grid.size()):
+		if grid[index] == -2 or grid[index] == -3 or _is_special(grid[index]):
+			continue
+		var choices: Array[int] = [0, 1, 2, 3, 4, 5]
+		if index % cols >= 2 and _is_ordinary(grid[index - 1]) and grid[index - 1] == grid[index - 2]:
+			choices.erase(grid[index - 1])
+		if index >= cols * 2 and _is_ordinary(grid[index - cols]) and grid[index - cols] == grid[index - cols * 2]:
+			choices.erase(grid[index - cols])
+		grid[index] = choices[_rng.randi_range(0, choices.size() - 1)]
+
+func _plant_move() -> bool:
+	# Same bounded fallback as the base, restricted to ordinary mineral slots.
+	var original := grid.duplicate()
+	for start in range(grid.size()):
+		for step in [1, cols]:
+			var triple: Array[int] = [start, start + step, start + step * 2]
+			if triple[2] >= grid.size() or (step == 1 and start % cols > cols - 3):
+				continue
+			if not _is_ordinary(original[triple[0]]) or not _is_ordinary(original[triple[1]]) or not _is_ordinary(original[triple[2]]):
+				continue
+			for destination in triple:
+				for source in _neighbors(destination):
+					if source in triple or not _is_ordinary(original[source]):
+						continue
+					for kind in range(GEM_KINDS):
+						for other in range(GEM_KINDS):
+							if other == kind:
+								continue
+							grid.assign(original)
+							for index in triple:
+								grid[index] = kind
+							grid[destination] = other
+							grid[source] = kind
+							if get_matches().is_empty() and not _find_move().is_empty():
+								return true
+	grid.assign(original)
+	return false
+
+func _is_special(kind: int) -> bool:
+	return kind == TORNADO_H or kind == TORNADO_V or kind == VOLCANO
+
+func _is_ordinary(kind: int) -> bool:
+	return kind >= 0 and kind < GEM_KINDS
+
+func _is_black_hole_pair(first: int, second: int) -> bool:
+	return (first == VOLCANO and (second == TORNADO_H or second == TORNADO_V)) or (second == VOLCANO and (first == TORNADO_H or first == TORNADO_V))
+
+func _power_name(kind: int) -> String:
+	match kind:
+		TORNADO_H: return "tornado_h"
+		TORNADO_V: return "tornado_v"
+	return "volcano"
