"""Download a commit-pinned probe package; validate everything before extraction."""
import argparse
import base64
import binascii
import hashlib
import io
import json
from pathlib import Path
import re
import stat
import urllib.request
import zipfile

REPO = 'Fuet-Goat/GnomiT-review-test'
PREFIX = 'discussions/codex/task-001/PROJECT_HUB/builds/grass-12-tuft-fill-probe-2026-10-22'
MAX_BYTES = 512 * 1024 * 1024
MAX_RECEIPT = 2 * 1024 * 1024

def require(condition, message):
    if not condition:
        raise ValueError(message)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def integer(value, label, minimum=0):
    require(type(value) is int and minimum <= value <= MAX_BYTES, 'Invalid '+label)
    return value

def checksum(value):
    require(isinstance(value, str) and re.fullmatch(r'[0-9a-f]{64}', value), 'Invalid SHA-256')
    return value

def safe_path(value):
    require(isinstance(value, str) and 0 < len(value) <= 240, 'Invalid package path')
    require('..' not in value and '\\' not in value and ':' not in value and not value.startswith('/'),
            'Unsafe package path: '+value)
    for piece in value.split('/'):
        require(re.fullmatch(r'[A-Za-z0-9_][A-Za-z0-9_.-]{0,127}', piece) and not piece.endswith('.'),
                'Non-conservative package name: '+value)
        require(not re.fullmatch(r'(?i)(CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9])', piece.split('.')[0]),
                'Reserved package name: '+value)
    return value

def unique_rows(rows, label):
    require(isinstance(rows, list) and 0 < len(rows) <= 10000, 'Invalid '+label+' list')
    names = set()
    for row in rows:
        require(isinstance(row, dict), 'Invalid '+label+' record')
        name = safe_path(row['path']).casefold()
        require(name not in names, 'Duplicate '+label+' path: '+row['path'])
        names.add(name)
        integer(row['bytes'], label+' bytes')
        checksum(row['sha256'])
    require(not any('/'.join(name.split('/')[:i]) in names
                    for name in names for i in range(1, len(name.split('/')))), 'File/directory path conflict')
    return names

def no_duplicate_keys(pairs):
    result = {}
    for key, value in pairs:
        require(key not in result, 'Duplicate JSON key: '+key)
        result[key] = value
    return result

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        raise ValueError('Unexpected download redirect')

def fetch(base, relative, limit):
    url = base+'/'+safe_path(relative)
    request = urllib.request.Request(url, headers={'User-Agent': 'Verified-GnomiT-Probe-Downloader/1'})
    with urllib.request.build_opener(NoRedirect()).open(request, timeout=60) as response:
        require(response.status == 200 and response.geturl() == url, 'Unexpected download response')
        length = response.headers.get('Content-Length')
        require(length is None or (length.isdigit() and int(length) <= limit), 'Response exceeds declared size')
        data = response.read(limit+1)
    require(len(data) <= limit, 'Download exceeds declared size: '+relative)
    return data

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--commit', required=True, help='Exact 40-hex Git commit')
    parser.add_argument('--destination', required=True, help='Absolute NEW extraction directory')
    args = parser.parse_args()
    require(re.fullmatch(r'[0-9a-fA-F]{40}', args.commit), '--commit must be exactly 40 hex characters')
    commit = args.commit.lower()
    destination = Path(args.destination)
    require(destination.is_absolute() and '..' not in destination.parts, '--destination must be an absolute new path')
    require(not destination.exists() and not destination.is_symlink(), 'Destination already exists')
    base = 'https://raw.githubusercontent.com/'+REPO+'/'+commit+'/'+PREFIX
    receipt = json.loads(fetch(base, 'PACKAGE_RECEIPT.json.txt', MAX_RECEIPT).decode('utf-8'),
                         object_pairs_hook=no_duplicate_keys)
    require(isinstance(receipt, dict), 'Invalid package receipt')
    package, parts, files = receipt['zip'], receipt['parts'], receipt['files']
    require(isinstance(package, dict), 'Invalid ZIP receipt')
    require('/' not in safe_path(package['name']) and package['name'].endswith('.zip'), 'Invalid ZIP name')
    zip_bytes = integer(package['bytes'], 'ZIP bytes', 1)
    zip_sha = checksum(package['sha256'])
    entries = integer(package['entries'], 'ZIP entries', 1)
    part_names = unique_rows(parts, 'part')
    file_names = unique_rows(files, 'file')
    require(not (part_names & file_names), 'Part and ZIP file names overlap')
    require(len(files) == entries and sum(row['bytes'] for row in files) <= MAX_BYTES, 'Declared file count/size failed')
    require(any(row['path'] == 'MANIFEST.json' for row in files), 'Local MANIFEST.json is required')
    assembled = bytearray()
    for part in parts:
        size = integer(part['bytes'], 'part bytes', 1)
        chars = integer(part['base64_chars'], 'base64 characters', 1)
        require(chars % 4 == 0 and chars <= ((zip_bytes+2)//3)*4 and size in (chars, chars+1), 'Invalid base64 size')
        checksum(part['payload_sha256'])
        encoded = fetch(base, part['path'], size)
        require(len(encoded) == size and sha(encoded) == part['sha256'], 'Part bytes/hash mismatch: '+part['path'])
        payload = encoded[:-1] if encoded.endswith(b'\n') else encoded
        require(len(payload) == chars, 'Base64 character count mismatch')
        decoded = base64.b64decode(payload, validate=True)
        require(sha(decoded) == part['payload_sha256'], 'Decoded part hash mismatch')
        require(len(assembled)+len(decoded) <= zip_bytes, 'Decoded parts exceed declared ZIP size')
        assembled.extend(decoded)
    require(len(assembled) == zip_bytes and sha(assembled) == zip_sha, 'Reconstructed ZIP bytes/hash mismatch')
    declared = {row['path']: row for row in files}
    checked = {}
    with zipfile.ZipFile(io.BytesIO(assembled)) as archive:
        members = archive.infolist()
        require(len(members) == entries, 'ZIP entry count mismatch')
        seen = set()
        for member in members:
            name = safe_path(member.filename)
            require(member.orig_filename == member.filename, 'ZIP filename normalization is forbidden')
            require(name.casefold() not in seen, 'Duplicate ZIP entry: '+name)
            seen.add(name.casefold())
            require(name in declared and not member.is_dir(), 'Undeclared ZIP entry: '+name)
            mode = member.external_attr >> 16
            require(not stat.S_ISLNK(mode) and stat.S_IFMT(mode) in (0, stat.S_IFREG), 'Unsafe ZIP file type: '+name)
            require(not member.flag_bits & 1, 'Encrypted ZIP entry: '+name)
            row = declared[name]
            require(member.file_size == row['bytes'], 'ZIP file exceeds/mismatches declared size: '+name)
            with archive.open(member) as stream:
                data = stream.read(row['bytes']+1)
            require(len(data) == row['bytes'] and sha(data) == row['sha256'], 'ZIP file bytes/hash mismatch: '+name)
            require(binascii.crc32(data) & 0xffffffff == member.CRC, 'ZIP CRC mismatch: '+name)
            checked[name] = data
    require(set(checked) == set(declared), 'ZIP entries differ from receipt')
    destination.mkdir(parents=True, exist_ok=False)
    for name, data in checked.items():
        target = destination.joinpath(*name.split('/'))
        target.parent.mkdir(parents=True, exist_ok=True)
        with target.open('xb') as output:
            output.write(data)
    for name, row in declared.items():
        actual = destination.joinpath(*name.split('/')).read_bytes()
        require(len(actual) == row['bytes'] and sha(actual) == row['sha256'], 'Extracted file verification failed: '+name)
    print(json.dumps({'commit': commit, 'zip_sha256': zip_sha, 'file_count': len(checked)}, sort_keys=True))

if __name__ == '__main__':
    main()
