+def noise(rng,n,low,high):
+    hz=np.fft.rfftfreq(n,1/RATE)
+    signal=np.fft.irfft(np.fft.rfft(rng.normal(size=n))*(1-np.exp(-(hz/low)**4))*np.exp(-(hz/high)**4),n=n)
+    return signal/max(float(np.sqrt(np.mean(signal*signal))),1e-9)

+def material(kind,t,rng,ordinal):
