func _ready() -> void:
	# Prefer monitor 2 when present, without changing desktop display settings.
	if not embedded and DisplayServer.get_name() != "headless":
		var target_screen := mini(1, DisplayServer.get_screen_count() - 1)
		DisplayServer.window_set_current_screen(target_screen)
		var usable := DisplayServer.screen_get_usable_rect(target_screen)
		DisplayServer.window_set_position(usable.position + (usable.size - DisplayServer.window_get_size()) / 2)
	if not embedded:DisplayServer.window_set_title("GnomiT · Mine V3 · living world and guide")
	_build_lighting()
	_build_controls()
	var packed := load(MODEL_PATH) as PackedScene
	if packed == null:
		push_error("Native storybook model could not be loaded: " + MODEL_PATH)
		return
	model = packed.instantiate() as Node3D
	if model == null:
		push_error("The storybook model root must be Node3D.")
		return
	add_child(model)
	_configure_imported_lights()
	_configure_camera()
	_merge_transform_animations()
	_build_level_markers()
	_update_level_markers()
	_update_lantern_energy()
	model_ready = true
	print("GNOMIT_PREVIEW_READY " + JSON.stringify({
		"screen_index": DisplayServer.window_get_current_screen(),
		"screen_count": DisplayServer.get_screen_count(),
		"transform_tracks": merged_transform_tracks,
		"lanterns": lanterns.size(),
		"scope": "Native campaign map" if embedded else "ART PREVIEW",
		"level_markers": level_markers.size(),
		"missing_anchors": missing_anchors
	}))


func _configure_imported_lights() -> void:
	for candidate in model.find_children("*", "Light3D", true, false):
		var lamp := candidate as Light3D
		var light_name := String(lamp.name).to_lower()
		if "inner lantern bounce" in light_name:
			lamp.visible = true
			lamp.light_color = Color(1.0, 0.59, 0.27)
			lamp.light_energy = 0.22
			lamp.shadow_enabled = false
			(lamp as OmniLight3D).omni_range = 1.8
			continue
		# Blender area/sun lights are not suitable real-time intensity defaults.
		var is_lantern := lamp is OmniLight3D and ("lantern" in light_name or "lamp" in light_name)
		lamp.visible = is_lantern
		if not is_lantern:
			hidden_imported_lights += 1
			continue
		lamp.light_color = Color(1.0, 0.57, 0.20)
		lamp.light_energy = 1.0
		lamp.shadow_enabled = true
		lamp.shadow_bias = 0.04
		var point := lamp as OmniLight3D
		point.omni_range = 3.4
		point.omni_attenuation = 1.25
		point.light_size = 0.16
		lanterns.append(lamp)


func _configure_camera() -> void:
	for candidate in model.find_children("*", "Camera3D", true, false):
		camera = candidate as Camera3D
		break
	if camera == null:
		camera = Camera3D.new()
		camera.name = "Fallback camera"
		add_child(camera)
		camera.position = Vector3(8.0, 17.0, 15.0)
		camera.look_at(Vector3(0.0, 0.7, 0.0))
		camera.projection = Camera3D.PROJECTION_ORTHOGONAL
		camera.size = 15.0
		push_warning("Imported camera absent; fallback framing requires visual review.")
	camera.make_current()
	source_camera_transform = camera.global_transform
	var bounds := _model_bounds()
	var focus_distance := maxf(1.0, source_camera_transform.origin.distance_to(bounds.get_center()))
	# Keep the imported view axis. Orbit does not re-aim or zoom the scene.
	orbit_target = source_camera_transform.origin - source_camera_transform.basis.z * focus_distance


func _model_bounds() -> AABB:
	var bounds := AABB()
	var has_bounds := false
	for candidate in model.find_children("*", "MeshInstance3D", true, false):
		var instance := candidate as MeshInstance3D
		if instance.mesh == null:
			continue
		var world_bounds: AABB = instance.global_transform * instance.get_aabb()
		bounds = bounds.merge(world_bounds) if has_bounds else world_bounds
		has_bounds = true
	return bounds


func _track_changes(animation: Animation, track: int) -> bool:
	if animation.track_get_key_count(track) < 2:
		return false
	var first: Variant = animation.track_get_key_value(track, 0)
	for key in range(1, animation.track_get_key_count(track)):
		var value: Variant = animation.track_get_key_value(track, key)
		if first is Vector3 and value is Vector3:
			if not first.is_equal_approx(value):
				return true
		elif first is Quaternion and value is Quaternion:
			if not first.is_equal_approx(value):
				return true
		elif first != value:
			return true
	return false


func _merge_transform_animations() -> void:
	for candidate in model.find_children("*", "AnimationPlayer", true, false):
		var player := candidate as AnimationPlayer
		var combined := Animation.new()
		combined.length = 6.0
		combined.loop_mode = Animation.LOOP_LINEAR
		var paths_seen: Dictionary = {}
		for source_name in player.get_animation_list():
			if String(source_name).get_file().to_upper() == "RESET":
				continue
			var source := player.get_animation(source_name)
			animation_source_clips.append(String(source_name))
			# Blender's frame1 is exported at 1/24s rather than zero. Keep its
			# six-second interval intact and put all authored first poses at t=0.
			var clip_origin := INF
			for track in range(source.get_track_count()):
				if source.track_get_type(track) in TRANSFORM_TYPES and _track_changes(source, track):
					clip_origin = minf(clip_origin, source.track_get_key_time(track,0))
			if is_inf(clip_origin): continue
			animation_source_origins[String(source_name)] = clip_origin
			combined.length = maxf(combined.length, source.length - clip_origin)
			for track in range(source.get_track_count()):
				if not source.track_is_enabled(track) or source.track_get_type(track) not in TRANSFORM_TYPES:
					continue
				if not _track_changes(source, track):
					continue
				var identity := "%s|%d" % [source.track_get_path(track), source.track_get_type(track)]
				if paths_seen.has(identity):
					continue
				paths_seen[identity] = true
				var copied_track := combined.get_track_count()
				source.copy_track(track, combined)
				for key in range(combined.track_get_key_count(copied_track)):
					combined.track_set_key_time(copied_track,key,maxf(0.0,combined.track_get_key_time(copied_track,key)-clip_origin))
		if combined.get_track_count() == 0:
			player.stop()
			continue
		var library := AnimationLibrary.new()
		library.add_animation("living_world", combined)
		player.add_animation_library("preview", library)
		player.callback_mode_process = AnimationMixer.ANIMATION_CALLBACK_MODE_PROCESS_MANUAL
		player.play(MERGED_ANIMATION)
		player.advance(0.0)
		players.append(player)
		merged_transform_tracks += combined.get_track_count()

func _build_wordmark() -> void:
	if not ResourceLoader.exists(WORDMARK_PATH):
		push_error("Element wordmark asset missing: " + WORDMARK_PATH)
		return
	var texture := load(WORDMARK_PATH) as Texture2D
	if texture == null:
		return
	var pixels := texture.get_image()
	if pixels == null:
		return
	wordmark_used_rect = pixels.get_used_rect()
	if wordmark_used_rect.size.x == 0 or wordmark_used_rect.size.y == 0:
		push_error("Element wordmark has empty alpha.")
		return
	# Trim only transparent padding through a texture region. Original image
	# bytes stay unchanged; this is ordinary UI framing, not a bitmap edit.
	var framed := AtlasTexture.new()
	framed.atlas = texture
	framed.region = Rect2(wordmark_used_rect)
	wordmark = TextureRect.new()
	wordmark.name = "GnomiT elemental lettering"
	wordmark.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	wordmark.texture = framed
	wordmark.position = Vector2(24,55)
	wordmark.size = Vector2(384,100)
	wordmark.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	wordmark.mouse_filter = Control.MOUSE_FILTER_IGNORE
	controls_layer.add_child(wordmark)
	wordmark_ready = true

func _find_anchor(expected_name: String) -> Node3D:
	for candidate in model.find_children("*", "Node3D", true, false):
		var candidate_name := String(candidate.name).replace("_", "").to_lower()
		if candidate_name == expected_name.replace("_", "").to_lower():
			return candidate as Node3D
	return null


func _build_level_markers() -> void:
	for number: int in [1, 2]:
		var anchor_name := "Level_%d_Anchor" % number
		var anchor := _find_anchor(anchor_name)
		if anchor == null:
			missing_anchors.append(anchor_name)
			push_error("Numbered art-preview marker has no native GLB anchor: " + anchor_name)
			continue
		marker_anchors[number] = anchor
		var button := Button.new()
		button.name = "Level %d preview marker" % number
		button.text = str(number)
		button.size = MARKER_SIZE
		button.custom_minimum_size = MARKER_SIZE
		button.focus_mode = Control.FOCUS_NONE
		# The map owns the complete press/drag/release gesture. These visual buttons
		# cannot swallow a drag that starts on a number and accidentally open it.
		button.mouse_filter = Control.MOUSE_FILTER_IGNORE
		button.add_theme_font_size_override("font_size", 29)
		button.add_theme_color_override("font_color", Color("fff8de"))
		button.add_theme_color_override("font_outline_color", Color("4c492a"))
		button.add_theme_constant_override("outline_size", 2)
		var fill := Color("568c35") if number == 1 else Color("527e91")
		button.add_theme_stylebox_override("normal", _marker_style(fill, Color("e9b551")))
		controls_layer.add_child(button)
		level_markers[number] = button


func _update_level_markers() -> void:
	if camera == null:
		return
	for number: int in level_markers:
		var anchor: Node3D = marker_anchors[number]
		var button: Button = level_markers[number]
		button.visible = not camera.is_position_behind(anchor.global_position)
		# Anchors are explicitly authored beside the arrival points in Blender;
		# preserving their projection keeps numbers on the road during orbit.
		button.position = camera.unproject_position(anchor.global_position) - MARKER_SIZE * 0.5

func show_level_intro(number: int) -> bool:
	if not LEVEL_COPY.has(number) or not level_markers.has(number):
		return false
	intro_level = number
	intro_title.text = String(LEVEL_COPY[number][0])
	intro_body.text = String(LEVEL_COPY[number][1])
	intro_panel.show()
	pointer_active = false
	return true


func dismiss_intro() -> void:
	intro_panel.hide()
	intro_level = 0
	pointer_active = false

func _toggle_motion() -> void:
	set_motion(not motion)


func set_motion(enabled: bool) -> void:
	motion = enabled
	if motion_button != null:
		motion_button.text = "Motion: ON" if motion else "Motion: OFF"


func reset_view() -> void:
	orbit = 0.0
	pointer_active = false
	if camera != null:
		camera.global_transform = source_camera_transform
	_update_level_markers()


func _unhandled_input(event: InputEvent) -> void:
	if camera == null:
		return
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
		if event.pressed:
			_begin_pointer(event.position, -1)
		elif pointer_touch_index == -1:
			_end_pointer(event.position)
	elif event is InputEventMouseMotion and pointer_active and pointer_touch_index == -1:
		_move_pointer(event.position, event.relative)
	elif event is InputEventScreenTouch:
		if event.pressed and not pointer_active:
			_begin_pointer(event.position, event.index)
		elif not event.pressed and event.index == pointer_touch_index:
			_end_pointer(event.position)
	elif event is InputEventScreenDrag and pointer_active and event.index == pointer_touch_index:
		_move_pointer(event.position, event.relative)


func _marker_at(point: Vector2) -> int:
	for number: int in level_markers:
		var marker: Button = level_markers[number]
		var center := marker.position + MARKER_SIZE * 0.5
		if marker.visible and point.distance_to(center) <= MARKER_SIZE.x * 0.5:
			return number
	return 0


func _begin_pointer(point: Vector2, touch_index: int) -> void:
	if point.y < 65.0 or point.y > 895.0:
		return
	pointer_active = true
	pointer_touch_index = touch_index
	pointer_origin = point
	pointer_max_distance = 0.0
	pressed_level = _marker_at(point)


func _move_pointer(point: Vector2, relative: Vector2) -> void:
	pointer_max_distance = maxf(pointer_max_distance, pointer_origin.distance_to(point))
	if pointer_max_distance < TAP_DISTANCE:
		return
	orbit = clampf(orbit + relative.x * 0.0015, -0.12, 0.12)
	var rotation := Basis(Vector3.UP, orbit)
	var view := source_camera_transform
	view.origin = orbit_target + rotation * (source_camera_transform.origin - orbit_target)
	view.basis = rotation * source_camera_transform.basis
	camera.global_transform = view
	_update_level_markers()


func _end_pointer(point: Vector2) -> void:
	if not pointer_active:
		return
	pointer_max_distance = maxf(pointer_max_distance, pointer_origin.distance_to(point))
	var selected := pressed_level
	var is_short_press := pointer_max_distance < TAP_DISTANCE
	pointer_active = false
	pointer_touch_index = -1
	pressed_level = 0
	if is_short_press and selected != 0 and selected == _marker_at(point):
		show_level_intro(selected)


func _process(delta: float) -> void:
	if not model_ready:
		return
	_update_level_markers()
	if not motion:
		return
	age += delta
	for player in players:
		player.advance(delta)
	_update_lantern_energy()


func _update_lantern_energy() -> void:
	# glTF omits Blender light-DATA F-curves. Preserve its six-second energy keys,
	# normalized around 1.0; smoothstep approximates the source AUTO_CLAMPED easing.
	var loop_time := fposmod(age, 6.0)
	var key := mini(int(floor(loop_time)), 5)
	var blend := smoothstep(0.0, 1.0, loop_time - float(key))
	var energy := lerpf(LANTERN_WATTS[key], LANTERN_WATTS[key + 1], blend) / 24.0
	for lamp in lanterns:
		lamp.light_energy = energy


func get_transform_snapshot() -> Dictionary:
	var result: Dictionary = {}
	if model == null:
		return result
	result[String(model.get_path())] = model.transform
	for candidate in model.find_children("*", "Node3D", true, false):
		var node := candidate as Node3D
		result[String(node.get_path())] = node.transform
	return result


func get_lamp_snapshot() -> Dictionary:
	var result: Dictionary = {}
	for lamp in lanterns:
		result[String(lamp.get_path())] = lamp.light_energy
	return result


func get_asset_inventory() -> Dictionary:
	var materials: Dictionary = {}
	var textures: Dictionary = {}
	var mesh_count := 0
	var surface_count := 0
	var texture_references := 0
	var shader_materials := 0
	if model == null:
		return {}
	for candidate in model.find_children("*", "MeshInstance3D", true, false):
		var instance := candidate as MeshInstance3D
		if instance.mesh == null:
			continue
		mesh_count += 1
		for surface in range(instance.mesh.get_surface_count()):
			surface_count += 1
			var material := instance.get_active_material(surface)
			if material == null or materials.has(material.get_instance_id()):
				continue
			materials[material.get_instance_id()] = String(material.resource_name)
			if material is BaseMaterial3D:
				for slot in range(BaseMaterial3D.TEXTURE_MAX):
					var texture: Texture2D = material.get_texture(slot)
					if texture != null:
						textures[texture.get_instance_id()] = String(texture.resource_path)
						texture_references += 1
			elif material is ShaderMaterial:
				shader_materials += 1
	return {
		"mesh_instances": mesh_count,
		"mesh_surfaces": surface_count,
		"unique_materials": materials.size(),
		"unique_material_textures": textures.size(),
		"material_texture_references": texture_references,
		"shader_materials_not_texture_inspected": shader_materials,
		"texture_paths": textures.values(),
		"material_names": materials.values()
	}
