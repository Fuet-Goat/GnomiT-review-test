extends Control
## Teaching illustration only: never creates a Core, board, reward or sound.
signal phase_changed(text: String)

const Art = preload("res://scripts/story_art.gd")
const MAGMA = preload("res://assets/vfx/volcano/magma_pebble.png")
const FLAME = preload("res://assets/vfx/volcano/lava_flame.png")
const CYCLE_SECONDS := 5.8
const TARGET_CELLS := [0, 4, 12]

var page_index := 0
var reduced_motion := false
var elapsed := 0.0
var phase_text := ""
var _font: Font
var _window_focused := true

func _ready() -> void:
	mouse_filter = Control.MOUSE_FILTER_IGNORE
	_font = Art.font(800)
	_refresh_phase()
	set_process(not reduced_motion)

func configure(page: int, reduced: bool) -> void:
	page_index = clampi(page, 0, 1)
	reduced_motion = reduced
	elapsed = 0.0
	set_process(not reduced_motion)
	_refresh_phase()
	queue_redraw()

func _process(delta: float) -> void:
	if not is_visible_in_tree() or not _window_focused: return
	elapsed = fposmod(elapsed + delta, CYCLE_SECONDS)
	_refresh_phase()
	queue_redraw()

func _notification(what: int) -> void:
	if what == NOTIFICATION_APPLICATION_FOCUS_OUT: _window_focused = false
	elif what == NOTIFICATION_APPLICATION_FOCUS_IN: _window_focused = true

func _refresh_phase() -> void:
	var text := "Four in a row → Volcano" if page_index == 0 else "Volcano → 3–5 minerals"
	if not reduced_motion:
		if page_index == 0:
			if elapsed < 1.1: text = "Move this ore up one cell"
			elif elapsed < 1.8: text = "Swap two neighbouring minerals"
			elif elapsed < 2.6: text = "Four matching minerals in one line"
			else: text = "A Volcano is ready for your next move"
		else:
			if elapsed < 1.1: text = "Move the Volcano to a neighbouring cell"
			elif elapsed < 1.8: text = "Your move triggers the eruption"
			elif elapsed < 3.5: text = "Burning rocks fly to scattered minerals"
			else: text = "The Volcano clears 3–5 minerals"
	if text != phase_text:
		phase_text = text
		phase_changed.emit(text)

func get_snapshot() -> Dictionary:
	return {"page_index":page_index,"reduced_motion":reduced_motion,"elapsed":elapsed,
		"cycle_seconds":CYCLE_SECONDS,"phase_text":phase_text,"example_hit_count":3,
		"actual_target_range":[3,5],"sound_players":0,"game_state_writes":false}

func _cell(center: Vector2, step: float, highlight: bool = false) -> void:
	var box := StyleBoxFlat.new()
	box.bg_color = Color("244b53") if not highlight else Color("4c6542")
	box.border_color = Color("5c8582") if not highlight else Color("ffda8b")
	box.set_border_width_all(2 if not highlight else 3)
	box.set_corner_radius_all(12)
	draw_style_box(box,Rect2(center-Vector2.ONE*step*.46,Vector2.ONE*step*.92))

func _mineral(center: Vector2, kind: int, radius: float, alpha: float = 1.0) -> void:
	draw_texture_rect(Art.treasure(kind),Rect2(center-Vector2.ONE*radius,Vector2.ONE*radius*2.0),false,Color(1,1,1,alpha))

func _volcano(center: Vector2, radius: float, alpha: float = 1.0) -> void:
	# The real board's ore volume, orange fissure and elliptical crater remain
	# recognisable here. The small ore chip conveys its matching mineral.
	draw_circle(center+Vector2(0,radius*.1),radius*1.05,Color(.07,.15,.16,alpha),true,-1,true)
	draw_arc(center+Vector2(0,radius*.1),radius*1.02,0,TAU,40,Color(.88,.65,.25,alpha),2.3,true)
	_mineral(center+Vector2(0,-radius*.08),2,radius*.83,alpha)
	draw_polyline(PackedVector2Array([center+Vector2(-.18,-.55)*radius,center+Vector2(-.08,-.17)*radius,center+Vector2(.19,.08)*radius,center+Vector2(.10,.53)*radius]),Color(1,.39,.10,alpha),maxf(3,radius*.12),true)
	draw_set_transform(center+Vector2(-.02,-.68)*radius,0,Vector2(1,.38))
	draw_circle(Vector2.ZERO,radius*.39,Color(.18,.10,.10,alpha),true,-1,true)
	draw_arc(Vector2.ZERO,radius*.39,0,TAU,28,Color(1,.65,.23,alpha),maxf(2,radius*.10),true)
	draw_set_transform(Vector2.ZERO)
	var chip := center+Vector2(-.70,.66)*radius
	draw_circle(chip,radius*.34,Color(.04,.10,.12,alpha),true,-1,true)
	_mineral(chip,2,radius*.35,alpha)

func _arrow(start: Vector2, end: Vector2, color: Color, width: float = 4.0) -> void:
	var direction := (end-start).normalized()
	var side := direction.orthogonal()
	draw_line(start,end,color,width,true)
	draw_polyline(PackedVector2Array([end-direction*14+side*9,end,end-direction*14-side*9]),color,width,true)

func _touch_hint(point: Vector2) -> void:
	draw_circle(point,12,Color(.98,.95,.80,.16),true,-1,true)
	draw_arc(point,13,0,TAU,32,Color("fff2ce"),3,true)
	draw_circle(point,4,Color("fff2ce"),true,-1,true)

func _draw() -> void:
	if size.x <= 0 or size.y <= 0: return
	var s := minf(size.x/560.0,size.y/400.0)
	var offset := (size-Vector2(560,400)*s)*.5
	draw_set_transform(offset,0,Vector2.ONE*s)
	# The icon helpers use absolute coordinates and reset transforms, so the
	# teaching diagram is drawn once into a fixed logical canvas transform.
	# All coordinates below are mapped explicitly to preserve responsive size.
	draw_set_transform(Vector2.ZERO)
	if page_index == 0: _draw_creation(offset,s)
	else: _draw_eruption(offset,s)

func _point(value: Vector2, offset: Vector2, scale_value: float) -> Vector2:
	return offset+value*scale_value

func _draw_creation(offset: Vector2, s: float) -> void:
	if reduced_motion:
		for column in range(4):
			var p := _point(Vector2(136+column*96,105),offset,s)
			_cell(p,88*s,true);_mineral(p,2,34*s)
		_arrow(_point(Vector2(280,177),offset,s),_point(Vector2(280,226),offset,s),Color("ffd28a"),4*s)
		_volcano(_point(Vector2(280,299),offset,s),48*s)
		return
	var centers: Array[Vector2] = []
	for row in range(2):
		for column in range(4):
			centers.append(_point(Vector2(136+column*96,140+row*96),offset,s))
	var swap := smoothstep(1.1,1.8,elapsed)
	var condense := smoothstep(2.6,3.2,elapsed)
	var kinds := [2,2,3,2,4,5,2,0]
	for index in centers.size():
		_cell(centers[index],88*s,index<4 and elapsed>=1.8 and elapsed<3.2)
		if index == 2 or index == 6: continue
		if index<4 and elapsed>=2.6:
			if condense<1: _mineral(centers[index].lerp(centers[2],condense),2,34*s,1-condense)
		else: _mineral(centers[index],int(kinds[index]),34*s)
	if elapsed<2.6:
		_mineral(centers[6].lerp(centers[2],swap),2,34*s)
		_mineral(centers[2].lerp(centers[6],swap),3,34*s)
	else:
		_mineral(centers[6],3,34*s)
		if condense<1: _mineral(centers[2],2,34*s,1-condense)
		if condense>0: _volcano(centers[2],34*s,condense)
	if elapsed<1.8:
		_arrow(centers[6]+Vector2(0,-25*s),centers[2]+Vector2(0,28*s),Color("ffe0a1"),4*s)
		_touch_hint(centers[6].lerp(centers[2],swap)+Vector2(24,18)*s)

func _board_point(index: int, offset: Vector2, s: float) -> Vector2:
	return _point(Vector2(120+(index%5)*80,94+(index/5)*80),offset,s)

func _draw_eruption(offset: Vector2, s: float) -> void:
	var origin := _board_point(7,offset,s)
	var destination := _board_point(8,offset,s)
	var swap := smoothstep(1.1,1.8,elapsed)
	var source := origin.lerp(destination,swap) if not reduced_motion else destination
	for index in range(15):
		var p := _board_point(index,offset,s)
		var target := TARGET_CELLS.has(index)
		_cell(p,73*s,target and (reduced_motion or elapsed>=1.8))
		if index in [7,8]: continue
		if target and not reduced_motion and elapsed>=3.5:
			# Visible empty cells explain that hits remove minerals, not the board.
			continue
		_mineral(p,(index*3+index/5)%6,29*s)
	_mineral(destination.lerp(origin,swap) if not reduced_motion else origin,4,29*s)
	if reduced_motion or elapsed<3.5: _volcano(source,29*s)
	if not reduced_motion and elapsed<1.8:
		_arrow(origin+Vector2(24,-30)*s,destination+Vector2(-22,-30)*s,Color("ffe0a1"),4*s)
		_touch_hint(source+Vector2(18,21)*s)
	for hit in range(TARGET_CELLS.size()):
		var target := _board_point(TARGET_CELLS[hit],offset,s)
		if reduced_motion:
			_arrow(source.lerp(target,.25),source.lerp(target,.80),Color("ffbd6e"),3*s)
			_fireball(source.lerp(target,.55),s,hit*.3)
			continue
		var start := 1.95+hit*.16
		var flight := clampf((elapsed-start)/1.1,0,1)
		if elapsed>=start and flight<1:
			var p := source.lerp(target,flight)+Vector2(0,-sin(flight*PI)*90*s)
			_fireball(p,s,elapsed*8+hit)
		elif flight>=1 and elapsed<3.5:
			var fade := 1-clampf((elapsed-start-1.1)/.45,0,1)
			draw_circle(target,26*s,Color(1,.56,.13,fade*.38),true,-1,true)
			for spark in range(5):
				var a := float(spark)/5*TAU
				draw_circle(target+Vector2(cos(a),sin(a))*22*s,3.2*s,Color(1,.77,.3,fade),true,-1,true)
	if not reduced_motion and elapsed>=1.8 and elapsed<3.1:
		var flame_size := Vector2(58,96)*s
		draw_texture_rect(FLAME,Rect2(source-Vector2(flame_size.x*.5,flame_size.y*1.05),flame_size),false,Color(1,1,1,.8))

func _fireball(point: Vector2, s: float, phase: float) -> void:
	var radius := (13+sin(phase)*1.2)*s
	draw_circle(point,radius*1.4,Color(1,.30,.06,.16),true,-1,true)
	draw_texture_rect(MAGMA,Rect2(point-Vector2.ONE*radius,Vector2.ONE*radius*2),false)
