extends Node3D

var model: Node3D
var player: AnimationPlayer
var clip: StringName
var camera: Camera3D
var ui: CanvasLayer
var status: Label
var reduced := false
var detail := false
var capture_mode := false

func _ready() -> void:
	var environment := Environment.new()
	environment.background_mode = Environment.BG_COLOR
	environment.background_color = Color("85867f")
	environment.ambient_light_source = Environment.AMBIENT_SOURCE_COLOR
	environment.ambient_light_color = Color("ede9dc")
	environment.ambient_light_energy = 0.325
	environment.tonemap_mode = Environment.TONE_MAPPER_AGX
	environment.ssao_enabled = true
	environment.ssao_radius = 0.2
	environment.ssao_intensity = 1.0
	var world := WorldEnvironment.new()
	world.environment = environment
	add_child(world)
	var floor_mesh := MeshInstance3D.new()
	var plane := PlaneMesh.new()
	plane.size = Vector2(200,200)
	floor_mesh.mesh = plane
	floor_mesh.position.y = -0.012
	var floor_material := StandardMaterial3D.new()
	floor_material.albedo_color = Color("b4b1a4")
	floor_material.roughness = 0.85
	floor_mesh.material_override = floor_material
	add_child(floor_mesh)
	var key := DirectionalLight3D.new()
	key.rotation_degrees = Vector3(-38,-35,0)
	key.light_color = Color("fff1d8")
	key.light_energy = 0.8
	key.shadow_enabled = false
	key.light_angular_distance = 1.0
	key.directional_shadow_max_distance = 10
	add_child(key)
	var fill := OmniLight3D.new()
	fill.position = Vector3(2,2.4,3)
	fill.light_color = Color("deebff")
	fill.light_energy = 1.2
	fill.omni_range = 8
	add_child(fill)
	var rim := OmniLight3D.new()
	rim.position = Vector3(-1.7,2,-2)
	rim.light_color = Color("fff0d5")
	rim.light_energy = 0.6
	rim.omni_range = 7
	add_child(rim)
	camera = Camera3D.new()
	camera.projection = Camera3D.PROJECTION_ORTHOGONAL
	camera.size = 1.92
	camera.far = 100
	camera.position = Vector3(0,1.14,5)
	add_child(camera)
	camera.look_at(Vector3(0,0.8,0))
	camera.current = true
	model = load("res://assets/gnome_r3.glb").instantiate() as Node3D
	add_child(model)
	player = model.find_child("AnimationPlayer",true,false) as AnimationPlayer
	assert(player != null,"GLB must include animation player")
	for name in player.get_animation_list():
		if name != "RESET": clip = name
	assert(not clip.is_empty(),"GLB must include original greeting")
	player.get_animation(clip).loop_mode = Animation.LOOP_NONE
	make_ui()
	player.animation_finished.connect(func(_name): status.text = "Settled - replay when ready")
	if DisplayServer.get_screen_count()>1:
		DisplayServer.window_set_current_screen(1)
		DisplayServer.window_set_position(DisplayServer.screen_get_position(1)+Vector2i(70,70))
	play_greeting()

func make_ui() -> void:
	ui = CanvasLayer.new()
	add_child(ui)
	var panel := PanelContainer.new()
	panel.position = Vector2(12,10)
	panel.size = Vector2(456,58)
	ui.add_child(panel)
	var headings := VBoxContainer.new()
	panel.add_child(headings)
	var title := Label.new()
	title.text = "R3 - GODOT IMPORT PREVIEW"
	title.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	headings.add_child(title)
	status = Label.new()
	status.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	headings.add_child(status)
	var buttons := HBoxContainer.new()
	buttons.position = Vector2(18,618)
	ui.add_child(buttons)
	var replay := Button.new()
	replay.text = "Replay greeting"
	replay.custom_minimum_size = Vector2(146,38)
	replay.pressed.connect(play_greeting)
	buttons.add_child(replay)
	var motion := CheckButton.new()
	motion.text = "Reduced motion"
	motion.toggled.connect(set_reduced)
	buttons.add_child(motion)
	var close := CheckButton.new()
	close.text = "Detail"
	close.toggled.connect(set_detail)
	buttons.add_child(close)

func play_greeting() -> bool:
	if reduced or player.is_playing():return false
	player.play(clip)
	status.text = "Original six-second greeting"
	return true

func pose(time: float) -> void:
	player.play(clip)
	player.seek(time,true)
	player.advance(0)
	player.pause()

func set_reduced(value: bool) -> void:
	reduced = value
	if value:
		pose(0)
		status.text = "Reduced motion - calm resting pose"
	else:status.text = "Ready - replay greeting"

func set_detail(value: bool) -> void:
	detail = value
	if value:
		camera.size = 0.69
		camera.position = Vector3(-0.29,1.13,5.08)
		camera.look_at(Vector3(-0.29,0.79,0.08))
	else:
		camera.size = 1.92
		camera.position = Vector3(0,1.14,5)
		camera.look_at(Vector3(0,0.8,0))
