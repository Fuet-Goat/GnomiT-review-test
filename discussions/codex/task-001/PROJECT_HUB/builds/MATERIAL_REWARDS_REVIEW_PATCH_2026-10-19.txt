+		components.append(cells)

+func _power_component(origin: int, kind: int, excluded: Array) -> Dictionary:

+	# A reserved new reward is also a ray barrier. Core and the inherited

+	# quake_paths renderer must stop together instead of clearing past a gap.

+	var component := super._power_component(origin, kind, excluded)

+	var targets: Array[int] = []

+	if kind >= EARTHQUAKE and kind <= EARTHQUAKE_NE:

+		var rays: Array = []

+		for raw_ray in component["rays"]:

+			var ray: Array[int] = []

+			for index in raw_ray:

+				if excluded.has(index): break

+				ray.append(int(index))

+				if _is_ordinary(grid[index]) and not targets.has(index): targets.append(int(index))

+			rays.append(ray)

+		component["rays"] = rays

+	else:

+		for index in component["targets"]:

+			if _is_ordinary(grid[index]) and not excluded.has(index): targets.append(int(index))

+	targets.sort()

+	component["targets"] = targets

+	var indices: Array[int] = targets.duplicate()

+	if not excluded.has(origin): indices.append(origin)

+	indices.sort()

+	component["indices"] = indices

+	return component

+func _resolve_cascades(events: Array) -> void:

+func _safe_fill_choices(index: int) -> Array[int]:

+	var choices: Array[int] = []

+	for kind in range(GEM_KINDS):

+		var allowed := true

+			for start in range(-2, 1):

+				for offset in range(3):

+					var dx := start + offset if horizontal else 0

+					var dy := 0 if horizontal else start + offset

+					var y: int = index / cols + dy

+						break

+				if same:

+func _fall_segment(segment: Array[int], falls: Array, spawn_number: int) -> int:

+	var gems: Array = []

+	for position in range(segment.size() - 1, -1, -1):

+		var index := segment[position]

+			gems.append({"from": index, "kind": grid[index], "reward": reward_at(index), "instance": _material_ids[index]})

+		_rewards.erase(index)

+		grid[index] = -1

+		_material_ids[index] = 0

+	var write_position := segment.size() - 1

+	for gem in gems:

+		var destination := segment[write_position]

+		grid[destination] = int(gem["kind"])

+		_material_ids[destination] = int(gem["instance"])

+		if not gem["reward"].is_empty(): _rewards[destination] = gem["reward"].duplicate(true)

+		if int(gem["from"]) != destination: falls.append({"from": int(gem["from"]), "to": destination, "kind": int(gem["kind"])})

+		write_position -= 1

+	if not _no_spawn:

+		while write_position >= 0:

+			var destination := segment[write_position]

+			var kind := _rng.randi_range(0, GEM_KINDS - 1)

+			grid[destination] = kind

+			_material_ids[destination] = _new_material_id()

+			spawn_number += 1

+			falls.append({"from": -spawn_number, "to": destination, "kind": kind})

+			write_position -= 1

+func _find_move() -> Array:

+	var rewards := _reward_snapshot()

+	for index in range(grid.size()):

+		if grid[index] < 0: continue

+		for neighbor in [index + 1, index + cols]:

+			if not _can_swap(index, neighbor): continue

+			var first_special := _is_special(grid[index])

+			var second_special := _is_special(grid[neighbor])

+			if first_special and second_special:

+				if _registered_pair(index, neighbor): return [index, neighbor]

+				continue

+			if first_special or second_special:

+				if not reward_at(index if first_special else neighbor).is_empty(): return [index, neighbor]

+				continue

+			if grid[index] == grid[neighbor]: continue

+			var snapshot: Array[int] = _material_snapshot(grid, rewards)

+			var old := snapshot[index]

+			snapshot[index] = snapshot[neighbor]

+			snapshot[neighbor] = old

+			if not _initial_material_matches(snapshot, index, neighbor).is_empty(): return [index, neighbor]

+	return []

+

+func _make_playable() -> bool:

+	var before_instances := _material_ids.duplicate()

+	var playable := super._make_playable()

+	_material_ids.resize(grid.size())

+	for index in range(grid.size()):

+		if _is_ordinary(grid[index]): _material_ids[index] = _new_material_id()

+		elif grid[index] < 0: _material_ids[index] = 0

+		elif reward_at(index).is_empty():

+			_material_ids[index] = _new_material_id()

+		elif index < before_instances.size(): _material_ids[index] = before_instances[index]

+	_pending_before_instances = before_instances

+	_ensure_instances()

+	return playable

+

+func _event(event_type: String, before: Array[int], extra: Dictionary = {}) -> Dictionary:

+	var details := extra.duplicate(true)

+	details["after_instances"] = _material_ids.duplicate()

+TRY THE DEFAULT EXAMPLE

+The initial dropdown is "Move ore Volcano -> eruption + new Volcano".

+Grab the ore Volcano in row 3, column 4 and drag DOWN exactly one cell to the

+glowing neighbour in row 4, column 4. These are zero-based grid indices 17 -> 24.

+The old Volcano erupts once. Its ore material completes a four-match, which

+leaves a NEW Volcano on the board. The newborn does not erupt automatically.

+You can then choose when to drag that new power to an adjacent cell to use it.

+

+CONTROLS

+- Hold the left mouse button on a stone or power, drag towards one orthogonally

+  adjacent cell, and release. Each gesture exchanges at most one neighbour.

+  An invalid swap spends no move.

+- Show move highlights the fixture move initially and a current legal move

+  after play has begun.

+- Reset restores the selected fixture and its seed, cancelling old animation

+  and sound. It does not reset the original game or any campaign save.

+- The top dropdown switches examples. You can keep playing the resulting board

+  after the advertised first move, rather than watching a standalone animation.

+- Skip effect finishes presentation immediately with the same resolved Core

+  state and no additional move. It also stops the active effect audio.

+- Reduced motion uses the reduced presentation. Select it while the board is

+  idle. It changes presentation, not rewards, collected minerals or moves.

+- Painted Volcano changes the Volcano presentation while idle; it does not

+  change the affected cells or the reward rules.

+

+WHAT MAKES A POWER

+The SHAPE chooses the power. The mineral does not choose its power family.

+- Four in a straight line: Volcano, four-born.

+- Five or more in a straight line: Storm, five-born.

+- T or plus-shaped matching runs: Tornado, five-born.

+- L-shaped matching runs: Landslide, five-born, retaining its L orientation.

+For larger intersecting shapes, the existing lab priority is straight five,

+then T/plus, then L, then straight four. One connected matched component earns

+at most one power; independent components can each earn a power.

+

+MATERIAL AND ABILITY ARE SEPARATE

+Four rubies create a ruby Volcano. Four ore pieces create an ore Volcano.

+Both erupt as Volcanos, but keep different stored mineral materials.

+The small mineral chip and coloured base identify the stored material.

+The main icon identifies the ability. FIVE-TICK marks a five-born ability.

+Ruby, amethyst, ore, diamond, emerald and pearl materials are supported.

+Material, ability, birth tier, ability ID and material-instance ID are tracked

+separately through swaps, gravity and ordinary reshuffling.

+

+WAITING POWERS, CATALYSTS AND CASCADES

+- Ordinary four/five/T/L matches after falling can create powers too.

+- Every newborn waits until the player deliberately swaps that power.

+  Birth, gravity, passing turns and nearby powers never activate it.

+- An unmoved waiting power can contribute its stored mineral to an INITIAL

+  player-made colour match. It remains the same waiting ability and does not

+  erupt or get collected. A new reward uses an ordinary matched anchor.

+- Example: move ordinary ore to complete [ore, ore-Volcano, ore, ore].

+  The stationary Volcano stays; the ordinary four-match leaves another Volcano.

+- If you deliberately MOVE an old power into a four/five match, the old effect

+  fires once and the new reward survives. Its reserved anchor is protected.

+- Automatic cascade matching considers ordinary stones only. Waiting powers

+  break those runs: [ruby, ruby-Volcano, ruby] does not automatically clear,

+  activate the Volcano, or mint another power after falling.

+- A retained catalyst or reserved new-reward anchor collects zero mineral units.

+  A genuinely removed power can contribute at most one stored mineral unit,

+  never the original four/five stones again.

+

+DIRECT POWER PAIRS

+Deliberately swapping two registered powers takes priority over colour recipes.

+- Four-born + four-born: two Volcano eruptions in one move.

+- Four-born + five-born: the five-born effect first, then Volcano targets the

+  remaining eligible ordinary stones. Shared cells are not counted twice.

+- Five-born + five-born: ONE Black Hole, including two identical power families.

+  These must be two already existing powers swapped directly by the player.

+  Incidental adjacency or matching colours never causes Black Hole.

+- Black Hole swallows other waiting powers without triggering any of their

+  separate effects or another Black Hole. Its direct sweep preserves soil and

+  runes; later genuine ordinary cascades can affect terrain normally.

+	return {"grid":core.grid.duplicate(), "floor":core.floor.duplicate(), "runes":core.runes.duplicate(), "rewards":_all_rewards(core), "instances":core._material_ids.duplicate(), "paid_instances":core._paid_instances.duplicate(true), "paid_components":core._paid_components.duplicate(true), "recovery":core.recovery_observations.duplicate(true), "moves":core.moves_left, "collected":core.collected, "rng":core._rng.state, "won":core.won, "lost":core.lost}

+

+func _events(result: Dictionary, event_type: String) -> Array:

+	var output: Array = []

+	for event in result.get("events", []):

+		if event.get("type", "") == event_type:

+			output.append(event)

+	return output

+

+func _awards(result: Dictionary) -> Array:

+	var output: Array = []

+	for event in _events(result, "reward_birth"):

+		output.append_array(event.get("awards", []))

+	return output

+

+func _effects(result: Dictionary) -> Array:

+	var output: Array = []

+	for event in result.get("events", []):

+		if event.get("type", "") == "natural_power":

+			output.append(event)

+		elif event.get("type", "") == "material_turn":

+			output.append_array(event.get("components", []))

+	return output

+

+func _mask(core, cells: Array, material: int = 0) -> Array[int]:

+	var output: Array[int] = []

+	for point in cells:

+		var index := int(point[1]) * 7 + int(point[0])

+		core.grid[index] = material

+		output.append(index)

+	output.sort()

+	core.refresh_material_instances()

+	return output

+

+func _pre_swap(cells: Array, destination: int, donor: int, seed_value: int = 42, material: int = 0):

+	var core = _new_core(seed_value, material)

+	_mask(core, cells, material)

+	var displaced: int = core.grid[donor]

+	core.grid[donor] = material

+	core.grid[destination] = displaced

+	core.refresh_material_instances()

+	return core

+

+func _replay_and_ledger(core, before: Dictionary, result: Dictionary) -> Dictionary:

+	var grid: Array = before.grid.duplicate()

+	var rewards: Dictionary = before.rewards.duplicate(true)

+	var instances: Array = before.instances.duplicate()

+	var valid := true

+	var expected_collected: int = before.collected

+	var count_removed := 0

+	for event in result.get("events", []):

+		valid = valid and event.get("before", []) == grid and event.get("before_rewards", {}) == rewards and event.get("before_instances",[]) == instances

+		if event.get("type", "") in ["clear", "natural_power", "material_turn"]:

+			var visited := {}

+			for index in event.get("indices", []):

+				if visited.has(index) or index in event.get("reserved_anchors", []):

+					continue

+				visited[index] = true

+				var material := int(event.before[index])

+				if material < 0 or material >= 6:

+					material = int(event.get("before_rewards", {}).get(index, {}).get("mineral_kind", -1))

+				if material >= 0 and material < 6:

+					count_removed += 1

+					if material == core.collect_kind:

+						expected_collected += 1

+		grid = event.get("after", []).duplicate()

+		rewards = event.get("after_rewards", {}).duplicate(true)

+		instances = event.get("after_instances",[]).duplicate()

+	valid = valid and grid == core.grid and rewards == _all_rewards(core) and instances == core._material_ids

+	return {"replay":valid, "ledger":expected_collected == core.collected, "expected_goal":expected_collected, "actual_goal":core.collected, "removed":count_removed}
