"""Rebuild v006 into a NEW folder from pinned v005 + verified small delta.
No engine or project code is executed. Optional --base-zip avoids a 56 MB download.
python DOWNLOAD_GNOMIT.py --output NEW_FOLDER [--base-zip LOCAL_V005_ZIP]
"""
import argparse,base64,hashlib,io,json,stat,urllib.request,zipfile
from pathlib import Path,PurePosixPath
REPO='Fuet-Goat/GnomiT-review-test'
PREFIX='discussions/codex/task-001/PROJECT_HUB/builds/black-hole-visible-layers-v006-2026-10-21/'
BASE_COMMIT='ef419de05d05b01be507c769c681c94fbf0fda28'
BASE_PREFIX='discussions/codex/task-001/PROJECT_HUB/builds/black-hole-one-hit-v005-2026-10-20/'
BASE_SHA='62cae5621af122dc3efa52c447e3b3cd7ca27bf39895e899071ef38b61c40c3a'
BASE_BYTES=56274647
DELTA_SHA='299292618cc25c2c64327b52cffb580538054d7d30b20897dc6d388c391e1d7e'
DELTA_BYTES=1636070
def get(url):
    req=urllib.request.Request(url,headers={'User-Agent':'GnomiT-verified-data-handoff'})
    with urllib.request.urlopen(req,timeout=90) as response:return response.read()
def verify_zip(payload,expected_hash,expected_bytes):
    if len(payload)!=expected_bytes or hashlib.sha256(payload).hexdigest()!=expected_hash:raise ValueError('Archive hash/size mismatch')
    bundle=zipfile.ZipFile(io.BytesIO(payload))
    if bundle.testzip() is not None:raise ValueError('Archive CRC mismatch')
    if len(bundle.namelist())!=len(set(bundle.namelist())):raise ValueError('Duplicate paths')
    for item in bundle.infolist():
        path=PurePosixPath(item.filename)
        if path.is_absolute() or '..' in path.parts or chr(92) in item.filename or ':' in item.filename or stat.S_ISLNK(item.external_attr>>16):raise ValueError('Unsafe archive path')
    return bundle
def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output',type=Path,required=True)
    parser.add_argument('--base-zip',type=Path)
    args=parser.parse_args();dest=args.output.resolve()
    if dest.exists():raise ValueError('Output must be a new directory')
    head=json.loads(get('https://api.github.com/repos/'+REPO+'/git/ref/heads/codex/codex/task-001'))['object']['sha']
    raw='https://raw.githubusercontent.com/'+REPO+'/'+head+'/'+PREFIX
    delta_payload=base64.b64decode(get(raw+'VERIFIED_DELTA.base64.txt'),validate=False)
    delta=verify_zip(delta_payload,DELTA_SHA,DELTA_BYTES)
    base_payload=args.base_zip.read_bytes() if args.base_zip else base64.b64decode(get('https://raw.githubusercontent.com/'+REPO+'/'+BASE_COMMIT+'/'+BASE_PREFIX+'NATIVE_PREVIEW_PACKAGE.base64.txt'))
    base=verify_zip(base_payload,BASE_SHA,BASE_BYTES)
    manifest=json.loads(delta.read('MANIFEST.json'))
    if len(manifest['files'])!=len({row['path'] for row in manifest['files']}):raise ValueError('Duplicate manifest paths')
    expected={row['path'] for row in manifest['files']}|{'MANIFEST.json'}
    if not set(delta.namelist())<=expected:raise ValueError('Unexpected delta paths')
    assembled={}
    for row in manifest['files']:
        name=row['path']
        data=delta.read(name) if name in delta.namelist() else base.read(name)
        if len(data)!=row['bytes'] or hashlib.sha256(data).hexdigest()!=row['sha256']:raise ValueError('Assembled file mismatch:'+name)
        assembled[name]=data
    dest.mkdir(parents=True,exist_ok=False)
    for name,data in assembled.items():
        target=dest/name;target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(data)
    (dest/'MANIFEST.json').write_bytes(delta.read('MANIFEST.json'))
    proof={'status':'PASS','source_commit':head,'base_commit':BASE_COMMIT,'base_sha256':BASE_SHA,'delta_sha256':DELTA_SHA,'verified_project_files':len(assembled),'no_engine_or_project_code_executed':True}
    (dest/'DOWNLOAD_VERIFIED.json').write_text(json.dumps(proof,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(proof,indent=2))
if __name__=='__main__':main()
