+"""Reproducible local foley, using the accepted stone break plus soft synthesis.
+
+No downloaded or provider-generated recordings. NumPy + stdlib only. Source
+stone audio is read without modification; all new sounds are separate files.
+"""
+from pathlib import Path
+import hashlib
+import json
+import wave
+import numpy as np
+
+ROOT = Path(__file__).resolve().parents[1]
+RATE = 44100
+RNG = np.random.default_rng(17102026)
+
+
+def noise(n, low, high):
+    raw = RNG.normal(size=n)
+    hz = np.fft.rfftfreq(n, 1 / RATE)
+    mask = (1 - np.exp(-(hz / low) ** 4)) * np.exp(-(hz / high) ** 4)
+    band = np.fft.irfft(np.fft.rfft(raw) * mask, n=n)
+    return band / max(np.sqrt(np.mean(band ** 2)), 1e-9)
+
+
+def envelope(n, attack, decay):
+    t = np.arange(n) / RATE
+    env = (1 - np.exp(-t / attack)) * np.exp(-t / decay)
+    # Smooth end, including exact zero endpoints, to avoid clicky cutoffs.
+    fade = min(int(RATE * .075), n // 3)
+    env[-fade:] *= np.linspace(1, 0, fade) ** 2
+    env[0] = env[-1] = 0
+    return env
+
+
+def write_sound(name, x, peak):
+    x = np.tanh(x * .65)
+    x *= peak / max(float(np.max(np.abs(x))), 1e-9)
+    x[0] = x[-1] = 0
+    pcm = np.round(x * 32767).astype('<i2')
+    path = ROOT / 'assets/sfx/volcano' / name
+    path.parent.mkdir(parents=True, exist_ok=True)
+    with wave.open(str(path), 'wb') as out:
+        out.setparams((1, 2, RATE, len(pcm), 'NONE', 'not compressed'))
+        out.writeframes(pcm.tobytes())
+    return {
+        'path': path.relative_to(ROOT).as_posix(), 'bytes': path.stat().st_size,
+        'sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
+        'seconds': len(x) / RATE, 'sample_rate': RATE, 'channels': 1,
+        'peak_dbfs': float(20 * np.log10(np.max(np.abs(pcm)) / 32768)),
+        'rms_dbfs': float(20 * np.log10(np.sqrt(np.mean((pcm / 32768.) ** 2)))),
+        'last_10ms_rms': float(np.sqrt(np.mean((pcm[-441:] / 32768.) ** 2))),
+    }
+
+
+def main():
+    source = ROOT / 'assets/sfx/stone_break_full_v1.wav'
+    with wave.open(str(source), 'rb') as inp:
+        assert inp.getparams()[:3] == (1, 2, RATE)
+        stone = np.frombuffer(inp.readframes(inp.getnframes()), '<i2').astype(float) / 32768
+    files = []
+
+    n = round(.92 * RATE)
+    t = np.arange(n) / RATE
+    # Warm rising pressure, airy release and a soft low thump, without a sharp
+    # explosion transient or piercing tonal chime.
+    rumble = noise(n, 45, 720) * (1 + .17 * np.sin(2 * np.pi * 17 * t))
+    breath = noise(n, 320, 2900)
+    swell = (1 - np.exp(-t / .045)) * np.exp(-t / .24)
+    low = np.sin(2 * np.pi * (85 * t - 25 * t ** 2)) * np.exp(-t / .17)
+    x = (.34 * rumble + .16 * breath) * swell + .19 * low
+    x *= envelope(n, .009, .85)
+    files.append(write_sound('eruption_soft_v1.wav', x, .67))
+
+    n = round(.30 * RATE)
+    t = np.arange(n) / RATE
+    # Small hot pebble crossing the air; a broad breath rather than a whistle.
+    x = (.26 * noise(n, 230, 2800) + .09 * noise(n, 80, 650))
+    x *= np.sin(np.pi * t / .30) ** 1.6
+    x *= envelope(n, .008, .70)
+    files.append(write_sound('lava_flight_soft_v1.wav', x, .39))
+
+    n = round(.72 * RATE)
+    t = np.arange(n) / RATE
+    # Keep the body of the user's chosen break, softened above 4.8 kHz. This
+    # is not just a tiny click: gritty fragments continue under a fire tail.
+    stone_part = stone[:n].copy()
+    hz = np.fft.rfftfreq(n, 1 / RATE)
+    stone_part = np.fft.irfft(np.fft.rfft(stone_part) * np.exp(-(hz / 4800) ** 4), n=n)
+    stone_part /= max(float(np.sqrt(np.mean(stone_part ** 2))), 1e-9)
+    body = .31 * stone_part * envelope(n, .003, .40)
+    thud = .17 * np.sin(2 * np.pi * (155 * t - 55 * t ** 2)) * envelope(n, .002, .075)

+        crackles[start:start+length] += strength * noise(length, 600, 4200) * envelope(length,.001,.008)
+    x = body + thud + fire + crackles
+    # Entire fragment body and fire finish smoothly, with no rough final snap.
+    x[-round(.12 * RATE):] *= np.linspace(1, 0, round(.12 * RATE)) ** 2

+        'seed': 17102026, 'source': source.relative_to(ROOT).as_posix(),
+        'source_sha256': hashlib.sha256(source.read_bytes()).hexdigest(),
+        'source_modified': False, 'human_listening_acceptance': 'PENDING', 'files': files,
+    }
+    target = ROOT / 'verification/volcano_audio_assets.json'
+    target.write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8')
+    print(json.dumps(report, indent=2))
+
+
+if __name__ == '__main__':
+    main()

+"""Extract header-described integer PCM from AVI/AVIX movies without a codec tool.
+
+Usage: python tools/extract_movie_audio.py movie.avi audio.wav --stats audio.json
+Use --stream N when the AVI declares more than one audio stream. Existing output
+files are never replaced. Only the selected stream's movi/rec audio is extracted;
+AVI stream start times, edit lists, and inter-stream mixing are not synthesized.
+
+Supported: little-endian PCM (8/16/24/32-bit), including PCM WAVEFORMATEXTENSIBLE.
+The WAV uses a conventional PCM header and preserves the original sample bytes;
+an extensible source's channel mask/valid-bit metadata remains in the JSON report.
+Compressed audio, floating-point audio, RF64 output, and damaged media are rejected.
+A complete final idx1 extending past an under-declared outer RIFF size is accepted
+with an explicit warning; media chunk boundaries remain strictly checked.
+Container references: https://learn.microsoft.com/en-us/windows/win32/directshow/avi-riff-file-reference
+"""
+
+from __future__ import annotations

+import argparse
+from dataclasses import asdict, dataclass, field
+import hashlib
+import json
+import math
+from pathlib import Path
+import struct
+import sys
+from typing import BinaryIO, Iterator
+import wave

+
+class AudioExtractionError(ValueError):
+    """Invalid AVI, unavailable audio, or unsupported audio format."""
+
+
+@dataclass(frozen=True)
+class Chunk:
+    tag: bytes
+    offset: int
+    size: int
+    list_type: bytes | None = None
+
+    @property
+    def end(self) -> int:
+        return self.offset + self.size
+
+
+@dataclass(frozen=True)
+class AudioStream:
+    index: int
+    scale: int
+    rate: int
+    start: int
+    length: int
+    sample_size: int
+    format_tag: int
+    channels: int
+    sample_rate: int
+    byte_rate: int
+    block_align: int
+    bits_per_sample: int
+    format_bytes: bytes = field(repr=False)
+
+    def metadata(self) -> dict:
+        return {key: value for key, value in asdict(self).items()
+                if key != "format_bytes"}

+@dataclass(frozen=True)
+class PCMFormat:
+    channels: int
+    sample_rate: int
+    sample_width: int
+    bits_per_sample: int
+    valid_bits_per_sample: int
+    block_align: int
+    channel_mask: int | None

+@dataclass(frozen=True)
+class AVIInfo:
+    source_bytes: int
+    segments: tuple[str, ...]
+    audio_streams: tuple[AudioStream, ...]
+    audio_chunks: tuple[tuple[int, Chunk], ...]
+    container_warnings: tuple[str, ...]

+def _read_at(source: BinaryIO, offset: int, size: int) -> bytes:
+    source.seek(offset)
+    data = source.read(size)
+    if len(data) != size:
+        raise AudioExtractionError(f"Truncated AVI at byte {offset}: need {size} bytes")
+    return data
+
+
+def iter_chunks(source: BinaryIO, start: int, end: int) -> Iterator[Chunk]:
+    """Yield direct RIFF children, checking lengths and word-alignment padding."""
+    cursor = start
+    while cursor < end:
+        if end - cursor < 8:
+            raise AudioExtractionError(f"Incomplete RIFF chunk header at byte {cursor}")
+        tag, size = struct.unpack("<4sI", _read_at(source, cursor, 8))
+        offset = cursor + 8
+        padded_end = offset + size + (size & 1)
+        if padded_end > end:
+            raise AudioExtractionError(f"Chunk {tag!r} at byte {cursor} exceeds its container")
+        kind = None
+        if tag in (b"LIST", b"RIFF"):
+            if size < 4:
+                raise AudioExtractionError(f"Short {tag!r} container at byte {cursor}")
+            kind = _read_at(source, offset, 4)
+        yield Chunk(tag, offset, size, kind)
+        cursor = padded_end
+
+
+def _children(source: BinaryIO, chunk: Chunk) -> Iterator[Chunk]:
+    return iter_chunks(source, chunk.offset + 4, chunk.end)
+
+
+def _segments(source: BinaryIO, size: int) -> tuple[list[Chunk], tuple[str, ...]]:
+    try:
+        return list(iter_chunks(source, 0, size)), ()
+    except AudioExtractionError as original_error:
+        # Godot's recorded AVI can under-declare only the outer size: its final
+        # idx1 is complete, but the RIFF size lands within that index. Validate
+        # every direct child against actual EOF before accepting this one case.
+        if size < 12:
+            raise
+        tag, declared_size, kind = struct.unpack("<4sI4s", _read_at(source, 0, 12))
+        declared_end = declared_size + 8
+        if tag != b"RIFF" or kind != b"AVI " or not 12 <= declared_end < size:
+            raise
+        try:
+            children = list(iter_chunks(source, 12, size))
+        except AudioExtractionError:
+            raise original_error
+        if (not children or children[-1].tag != b"idx1"
+                or children[-1].end != size
+                or not children[-1].offset - 8 < declared_end < children[-1].end
+                or any(child.end > declared_end or child.tag == b"RIFF" for child in children[:-1])):
+            raise original_error
+        warning = (f"Outer RIFF size ends {size - declared_end} bytes before EOF inside "
+                   "a complete final idx1; index ignored, media boundaries validated")
+        return [Chunk(b"RIFF", 8, size - 8, b"AVI ")], (warning,)
+
+
+def _audio_streams(source: BinaryIO, hdrl: Chunk) -> tuple[AudioStream, ...]:
+    streams = []
+    stream_lists = [chunk for chunk in _children(source, hdrl)
+                    if chunk.tag == b"LIST" and chunk.list_type == b"strl"]
+    for index, stream_list in enumerate(stream_lists):
+        chunks = list(_children(source, stream_list))
+        headers = [chunk for chunk in chunks if chunk.tag == b"strh"]
+        if len(headers) != 1 or headers[0].size < 4:
+            raise AudioExtractionError(f"Stream {index} needs one valid strh header")
+        header = _read_at(source, headers[0].offset, headers[0].size)
+        if header[:4] != b"auds":
+            continue
+        # Godot writes the 48-byte variant without the optional rcFrame tail.
+        if len(header) < 48:
+            raise AudioExtractionError(f"Audio stream {index} has a short strh header")
+        formats = [chunk for chunk in chunks if chunk.tag == b"strf"]
+        if len(formats) != 1 or formats[0].size < 16:
+            raise AudioExtractionError(f"Audio stream {index} needs one valid strf header")
+        fmt = _read_at(source, formats[0].offset, formats[0].size)
+        tag, channels, sample_rate, byte_rate, align, bits = struct.unpack_from("<HHIIHH", fmt)
+        scale, rate, start, length = struct.unpack_from("<IIII", header, 20)
+        sample_size = struct.unpack_from("<I", header, 44)[0]
+        streams.append(AudioStream(index, scale, rate, start, length, sample_size,
+                                   tag, channels, sample_rate, byte_rate, align, bits, fmt))
+    return tuple(streams)

+def _movi_audio(source: BinaryIO, container: Chunk, in_movi: bool = False,
+                depth: int = 0) -> Iterator[tuple[int, Chunk]]:
+    if depth > 64:
+        raise AudioExtractionError("RIFF nesting exceeds 64 containers")
+    for chunk in _children(source, container):
+        if chunk.tag == b"LIST":
+            yield from _movi_audio(source, chunk,
+                                   in_movi or chunk.list_type == b"movi", depth + 1)
+        elif in_movi and chunk.tag[2:] == b"wb":
+            prefix = chunk.tag[:2]
+            if len(prefix) != 2 or any(value not in b"0123456789ABCDEFabcdef" for value in prefix):
+                raise AudioExtractionError(f"Unsupported audio stream chunk ID {chunk.tag!r}")
+            # AVI's MAKEAVICKID/StreamFromFOURCC use two hexadecimal digits.
+            yield int(prefix, 16), chunk

+def inspect_avi(path: str | Path) -> AVIInfo:

+        size = source.seek(0, 2)

+        if not segments or segments[0].tag != b"RIFF" or segments[0].list_type != b"AVI ":
+            raise AudioExtractionError("Input must begin with a RIFF AVI segment")
+        for index, segment in enumerate(segments):
+            if segment.tag != b"RIFF" or segment.list_type not in (b"AVI ", b"AVIX"):
+                raise AudioExtractionError("Expected only RIFF AVI/AVIX segments at file level")
+            if index and segment.list_type != b"AVIX":
+                raise AudioExtractionError("Only AVIX continuation segments may follow the first AVI")
+        headers = [chunk for chunk in _children(source, segments[0])

+            raise AudioExtractionError("AVI needs exactly one hdrl list in its first segment")

+        unknown = sorted({index for index, _ in audio} - known)
+        if unknown:
+            raise AudioExtractionError(f"Audio chunks have no matching auds stream header: {unknown}")
+        return AVIInfo(size, tuple(segment.list_type.decode("ascii").strip()

+def pcm_format(stream: AudioStream) -> PCMFormat:
+    """Validate the selected WAVEFORMAT(EX/TENSIBLE); never guess a codec."""
+    fmt = stream.format_bytes
+    valid_bits = stream.bits_per_sample
+    mask = None
+    if len(fmt) == 17:
+        raise AudioExtractionError("Incomplete WAVEFORMATEX extension length")
+    if len(fmt) >= 18:
+        extra = struct.unpack_from("<H", fmt, 16)[0]
+        if 18 + extra > len(fmt):
+            raise AudioExtractionError("WAVEFORMATEX extension exceeds strf size")
+    if stream.format_tag == 0xFFFE:
+        if len(fmt) < 40 or extra < 22:
+            raise AudioExtractionError("Incomplete WAVEFORMATEXTENSIBLE PCM header")
+        valid_bits, mask = struct.unpack_from("<HI", fmt, 18)
+        pcm_guid = bytes.fromhex("0100000000001000800000aa00389b71")
+        if fmt[24:40] != pcm_guid:
+            raise AudioExtractionError("WAVEFORMATEXTENSIBLE subtype is not integer PCM")
+    elif stream.format_tag != 1:
+        raise AudioExtractionError(
+            f"Unsupported audio format tag 0x{stream.format_tag:04x}; integer PCM is required")
+    bits = stream.bits_per_sample
+    if bits not in (8, 16, 24, 32):
+        raise AudioExtractionError(f"Unsupported PCM container width: {bits} bits")
+    if not 1 <= valid_bits <= bits:
+        raise AudioExtractionError(f"Invalid PCM valid-bit count: {valid_bits}")
+    if not 1 <= stream.channels <= 65535 or stream.sample_rate <= 0:
+        raise AudioExtractionError("PCM channel count and sample rate must be positive")
+    width = bits // 8
+    align = width * stream.channels
+    if stream.block_align != align or stream.byte_rate != stream.sample_rate * align:
+        raise AudioExtractionError("PCM block alignment/byte rate disagrees with its strf format")
+    return PCMFormat(stream.channels, stream.sample_rate, width, bits, valid_bits, align, mask)
+

+def decode_pcm(data: bytes, fmt: PCMFormat) -> Iterator[int]:
+    """Decode interleaved little-endian samples to signed integer valid bits."""
+    if len(data) % fmt.block_align:
+        raise AudioExtractionError("PCM payload ends in a partial sample frame")
+    width = fmt.sample_width
+    shift = fmt.bits_per_sample - fmt.valid_bits_per_sample
+    for offset in range(0, len(data), width):
+        raw = data[offset:offset + width]
+        value = raw[0] - 128 if width == 1 else int.from_bytes(raw, "little", signed=True)
+        if shift and value & ((1 << shift) - 1):
+            raise AudioExtractionError("PCM sample has nonzero bits outside its declared valid bits")
+        yield value >> shift
+

+class PCMStats:
+    """Streaming exact sample counts and sums; floats only for derived levels."""
+
+    def __init__(self, fmt: PCMFormat):
+        self.fmt = fmt
+        self.counts = [0] * fmt.channels
+        self.nonzero = [0] * fmt.channels
+        self.clipped = [0] * fmt.channels
+        self.peaks = [0] * fmt.channels
+        self.squares = [0] * fmt.channels
+        self.sums = [0] * fmt.channels
+
+    def update(self, data: bytes) -> None:
+        floor = -(1 << (self.fmt.valid_bits_per_sample - 1))
+        ceiling = -floor - 1
+        for index, sample in enumerate(decode_pcm(data, self.fmt)):
+            channel = index % self.fmt.channels
+            self.counts[channel] += 1
+            self.nonzero[channel] += sample != 0
+            self.clipped[channel] += sample in (floor, ceiling)
+            self.peaks[channel] = max(self.peaks[channel], abs(sample))
+            self.squares[channel] += sample * sample
+            self.sums[channel] += sample
+
+    def report(self) -> dict:
+        full_scale = 1 << (self.fmt.valid_bits_per_sample - 1)
+
+        def levels(count: int, nonzero: int, clipped: int, peak: int,
+                   squares: int, sample_sum: int) -> dict:
+            rms = math.sqrt(squares / count) / full_scale if count else 0.0
+            return {"sample_count": count, "nonzero_samples": nonzero,
+                    "clipped_samples": clipped, "peak_integer": peak,
+                    "peak_normalized": peak / full_scale, "rms_normalized": rms,
+                    "rms_dbfs": 20 * math.log10(rms) if rms else None,
+                    "mean_normalized": sample_sum / count / full_scale if count else 0.0}
+
+        channels = [levels(*values) for values in zip(self.counts, self.nonzero,
+                    self.clipped, self.peaks, self.squares, self.sums)]
+        combined = levels(sum(self.counts), sum(self.nonzero), sum(self.clipped),
+                          max(self.peaks), sum(self.squares), sum(self.sums))
+        return {**combined, "per_channel": channels}
+
+
+def extract_audio(source_path: str | Path, output_path: str | Path,
+                  stats_path: str | Path | None = None, stream_index: int | None = None) -> dict:
+    """Write one selected stream as PCM WAV and return measurements of its bytes."""
+    source_path, output_path = Path(source_path).resolve(), Path(output_path).resolve()
+    stats_path = Path(stats_path).resolve() if stats_path is not None else None
+    destinations = [output_path] + ([stats_path] if stats_path is not None else [])
+    if len(set(destinations + [source_path])) != len(destinations) + 1:
+        raise AudioExtractionError("Input AVI, output WAV, and output JSON must be distinct paths")
+    for path in destinations:
+        if path.exists():
+            raise AudioExtractionError(f"Refusing to replace existing output: {path}")
+        if not path.parent.is_dir():
+            raise AudioExtractionError(f"Output directory does not exist: {path.parent}")
+    original = source_path.stat()
+    info = inspect_avi(source_path)
+    streams = {stream.index: stream for stream in info.audio_streams}
+    if not streams:
+        raise AudioExtractionError("AVI declares no audio stream")
+    if stream_index is None:
+        if len(streams) != 1:
+            raise AudioExtractionError(f"Multiple audio streams {sorted(streams)}; select one with --stream")
+        stream_index = next(iter(streams))
+    if stream_index not in streams:
+        raise AudioExtractionError(f"No audio stream {stream_index}; available: {sorted(streams)}")
+    stream = streams[stream_index]
+    fmt = pcm_format(stream)
+    chunks = [chunk for index, chunk in info.audio_chunks if index == stream_index]
+    total = sum(chunk.size for chunk in chunks)
+    if not total:
+        raise AudioExtractionError(f"Audio stream {stream_index} has no PCM payload")
+    if total % fmt.block_align:
+        raise AudioExtractionError("Combined audio payload ends in a partial PCM sample frame")
+    if total + (total & 1) > 0xFFFFFFFF - 36:
+        raise AudioExtractionError("Audio exceeds conventional RIFF WAV size; RF64 is unsupported")
+    stats = PCMStats(fmt)
+    pcm_hash = hashlib.sha256()
+    created = []
+    try:
+        with source_path.open("rb") as source, output_path.open("xb") as destination:
+            created.append(output_path)
+            with wave.open(destination, "wb") as wav:
+                wav.setparams((fmt.channels, fmt.sample_width, fmt.sample_rate,
+                               total // fmt.block_align, "NONE", "not compressed"))
+                carry = b""
+                for chunk in chunks:
+                    cursor, remaining = chunk.offset, chunk.size
+                    while remaining:
+                        amount = min(remaining, 1024 * 1024)
+                        data = carry + _read_at(source, cursor, amount)
+                        complete = len(data) - len(data) % fmt.block_align
+                        carry = data[complete:]
+                        data = data[:complete]
+                        stats.update(data)
+                        pcm_hash.update(data)
+                        wav.writeframesraw(data)
+                        cursor += amount
+                        remaining -= amount
+            if total & 1:
+                # wave omits RIFF's final word-alignment pad for odd data sizes.
+                # The pad belongs to RIFF, not to the data chunk/sample frames.
+                destination.seek(0, 2)
+                destination.write(b"\0")
+                destination.seek(4)
+                destination.write(struct.pack("<I", 36 + total + 1))
+        current = source_path.stat()
+        if (original.st_size, original.st_mtime_ns) != (current.st_size, current.st_mtime_ns):
+            raise AudioExtractionError("Input AVI changed during extraction")
+        with output_path.open("rb") as result:
+            wav_hash = hashlib.file_digest(result, "sha256").hexdigest()
+        report = {"source_avi": str(source_path), "output_wav": str(output_path),
+                  "source_bytes": info.source_bytes, "riff_segments": list(info.segments),
+                  "container_warnings": list(info.container_warnings),
+                  "audio_streams": [item.metadata() for item in info.audio_streams],
+                  "selected_stream": stream_index, "source_format": asdict(fmt),
+                  "wav_encoding": "integer PCM; original sample bytes; conventional PCM header",
+                  "audio_chunk_count": len(chunks), "pcm_bytes": total,
+                  "sample_frames": total // fmt.block_align,
+                  "duration_seconds": total / fmt.block_align / fmt.sample_rate,
+                  "pcm_sha256": pcm_hash.hexdigest(), "wav_sha256": wav_hash,
+                  "wav_bytes": output_path.stat().st_size, "statistics": stats.report()}
+        if stats_path is not None:
+            with stats_path.open("x", encoding="utf-8", newline="\n") as destination:
+                created.append(stats_path)
+                json.dump(report, destination, indent=2, allow_nan=False)
+                destination.write("\n")
+        return report
+    except BaseException:
+        for path in reversed(created):
+            path.unlink(missing_ok=True)
+        raise
+
+
+def main(argv: list[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
+    parser.add_argument("input_avi", type=Path)
+    parser.add_argument("output_wav", type=Path)
+    parser.add_argument("--stats", type=Path, help="Optional JSON measurements file")
+    parser.add_argument("--stream", type=int, help="Audio stream's actual zero-based strl index")
+    args = parser.parse_args(argv)
+    try:
+        report = extract_audio(args.input_avi, args.output_wav, args.stats, args.stream)
+    except (AudioExtractionError, OSError, wave.Error, struct.error) as error:
+        print(f"Audio extraction failed: {error}", file=sys.stderr)
+        return 2
+    print(json.dumps(report, indent=2, allow_nan=False))
+    return 0
