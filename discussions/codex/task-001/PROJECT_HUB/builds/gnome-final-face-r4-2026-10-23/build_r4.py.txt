"""Last bounded face correction from exact R3; cap and every other object frozen."""
from pathlib import Path
import sys,json
import bpy
from mathutils import Vector
ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT/'source'))
from evidence import sha,digest,fingerprint,material_record,render,detail
from face_r4 import build_face
SOURCE=ROOT.parent/'Gnome_Face_Cap_R3_2026-10-22/native/GNOME_R3_FACE_CAP_C.blend'
EXPECTED='72263fcd64dc756ee98b95073d166e453619b81c6420ef6b42230ece519988ac'
TAG='R4_FACE_ONLY_A'

def main():
    assert bpy.app.background and sha(SOURCE)==EXPECTED
    native=ROOT/'native'/f'GNOME_{TAG}.blend';assert not native.exists()
    bpy.ops.wm.open_mainfile(filepath=str(SOURCE));scene=bpy.context.scene
    asset=bpy.data.collections['R2_NEW_GNOME'];root=bpy.data.objects['R2_GnomeRoot']
    clay=bpy.data.materials['R1 Clay — form review only'];eye=bpy.data.materials['R1 Eye direction — blockout marker']
    assert not bpy.data.actions and not any(o.type=='ARMATURE' for o in bpy.data.objects)
    render('R3_BASELINE_FRONT.png');detail('R3_FACE_DETAIL.png')
    allowed={'R3_ContinuousFacialPlanes'}
    for s in (-1,1):allowed.update(p+str(s) for p in ['R3 inset eye ','R3 pupil ','R3 soft brow '])
    assert len(allowed)==7 and all(bpy.data.objects.get(n) for n in allowed)
    protected={o.name:fingerprint(o) for o in bpy.data.objects if o.name not in allowed}
    materials=material_record()
    for name in allowed:bpy.data.objects.remove(bpy.data.objects[name],do_unlink=True)
    face=build_face({'collection':asset,'root':root,'clay':clay,'eye':eye})
    names=[o.name for o in face['objects']]
    for obj in face['objects']:
        if obj.type=='MESH':
            bpy.ops.object.select_all(action='DESELECT');obj.select_set(True);bpy.context.view_layer.objects.active=obj
            bpy.ops.object.mode_set(mode='EDIT');bpy.ops.mesh.select_all(action='SELECT');bpy.ops.mesh.normals_make_consistent(inside=False);bpy.ops.object.mode_set(mode='OBJECT')
    bpy.context.view_layer.update()
    assert material_record()==materials
    assert set(o.name for o in bpy.data.objects)==set(protected)|set(names)
    for name,state in protected.items():assert fingerprint(bpy.data.objects[name])==state,'Protected object changed: '+name
    assert not bpy.data.actions and not any(o.type=='ARMATURE' for o in bpy.data.objects)
    text=bpy.data.texts.new('START_HERE_R4');text.write('Face-only expression correction. Eyelid opening, raised outer cheek transitions and fuller curved brows. Same eye globe dimensions. Cap, beard, moustache, smile, collar/body/hands/boots/root/studio/materials protected. No rig, animation, game integration. Human form decision pending.\n')
    scene.render.resolution_x=600;scene.render.resolution_y=840;bpy.context.preferences.filepaths.save_version=0
    assert 'FINISHED' in bpy.ops.wm.save_as_mainfile(filepath=str(native),check_existing=False)
    render('R4_FRONT.png')
    cam=scene.camera;oldtransform=(cam.location.copy(),cam.rotation_euler.copy(),cam.scale.copy());target=Vector((0,0,.8))
    cam.location=target+Vector((3,-4,.65));cam.rotation_euler=(target-cam.location).to_track_quat('-Z','Y').to_euler()
    render('R4_THREE_QUARTER.png');cam.location,cam.rotation_euler,cam.scale=oldtransform
    detail('R4_FACE_DETAIL.png')
    bpy.ops.wm.open_mainfile(filepath=str(native))
    for name,state in protected.items():assert fingerprint(bpy.data.objects[name])==state,'Protected object changed after reopen: '+name
    assert sha(SOURCE)==EXPECTED
    receipt={'status':'NATIVE_SAVED_REOPENED_RENDERED','source':str(SOURCE),'source_sha256_before':EXPECTED,'source_sha256_after':sha(SOURCE),
        'native':str(native),'native_sha256':sha(native),'replaced_objects':sorted(allowed),'new_objects':names,
        'protected_objects':len(protected),'protected_geometry_transforms_modifiers_material_slots':'EXACT_MATCH_BEFORE_AFTER_REOPEN',
        'protected_object_fingerprints':{n:digest(s) for n,s in protected.items()},'material_definitions_unchanged':True,
        'cap_beard_moustache_smile_collar_body_hands_boots_root':'UNCHANGED',
        'face':{k:v for k,v in face.items() if k not in ('objects','head')},
        'stage_dimensions':[600,840],'face_detail_dimensions':[640,640],
        'actions':len(bpy.data.actions),'armatures':sum(o.type=='ARMATURE' for o in bpy.data.objects),
        'human_form_acceptance':'PENDING','materials_rig_animation':'ON_HOLD','game_integration':False}
    (ROOT/'documentation/R4_RECEIPT.json').write_bytes((json.dumps(receipt,ensure_ascii=False,indent=2)+'\n').encode())
    print('R4_COMPLETE '+json.dumps({k:v for k,v in receipt.items() if k!='protected_object_fingerprints'}),flush=True)
if __name__=='__main__':main()
