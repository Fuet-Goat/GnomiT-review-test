+func _prepare_audio() -> void:
+	# Direct WAV loading also works before editor import has run.
+	for kind in STREAM_PATHS:
+		if streams.has(kind) or not FileAccess.file_exists(STREAM_PATHS[kind]): continue
+		var stream := AudioStreamWAV.load_from_file(STREAM_PATHS[kind])
+		if stream != null and stream.get_length() > 0.0: streams[kind] = stream
+	if not players.is_empty(): return
+	if AudioServer.get_bus_index(BUS_NAME) < 0:
+		AudioServer.add_bus()
+		var index := AudioServer.bus_count-1
+		AudioServer.set_bus_name(index,BUS_NAME)
+		AudioServer.set_bus_volume_db(index,-2.0)
+		var limiter := AudioEffectHardLimiter.new()
+		limiter.ceiling_db = -6.0
+		AudioServer.add_bus_effect(index,limiter)
+	for voice in VOICES:
+		var player := AudioStreamPlayer.new()
+		player.bus = BUS_NAME
+		add_child(player)
+		players.append(player)
+
+func asset_status() -> Dictionary:
+	_prepare_audio()
+	var missing: Array = []
+	var lengths: Dictionary = {}
+	for kind in STREAM_PATHS:
+		if not streams.has(kind): missing.append(STREAM_PATHS[kind])
+		else: lengths[kind] = streams[kind].get_length()
+	return {"ready":missing.is_empty(),"normal_ready":streams.has("normal"),"reduced_ready":streams.has("reduced"),"missing":missing,"lengths":lengths}
+
+func begin(reduced: bool) -> void:
+	stop_all()
+	_prepare_audio()
+	var selected := "reduced" if reduced else "normal"
+	if not streams.has(selected) or players.is_empty():
+		push_warning("Tornado audio asset unavailable; retaining baseline stone foley: " + STREAM_PATHS[selected])
+		return
+	_event_id += 1
+	var slot := _next_voice
+	_next_voice = (slot+1)%players.size()
+	var player: AudioStreamPlayer = players[slot]
+	player.stream = streams[selected]
+	player.pitch_scale = 1.0
+	player.volume_db = -15.0 if reduced else -7.0
+	player.play()
+	event_active = true
+	cue_log.append({"kind":"reduced" if reduced else "clip","reduced":reduced,"pitch_scale":player.pitch_scale,"volume_db":player.volume_db,"event_id":_event_id,"voice":slot,"source_path":STREAM_PATHS[selected],"length":player.stream.get_length()})
+
+func finish_event() -> void:
+	# Normal VFX completion leaves the single recording's short magic ending.
+	event_active = false
+
+func stop_all() -> void:
+	finish_event()
+	for player in players: player.stop()

+func _exit_tree() -> void:
+	stop_all()
