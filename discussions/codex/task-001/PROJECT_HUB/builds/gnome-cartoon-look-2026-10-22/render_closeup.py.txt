from pathlib import Path
import sys
import bpy
from mathutils import Vector

ROOT = Path(__file__).resolve().parents[1]

def closeup(scene, path):
    camera = scene.camera
    transform = (camera.location.copy(), camera.rotation_euler.copy(), camera.scale.copy())
    state = (camera.data.ortho_scale,
             scene.render.resolution_x, scene.render.resolution_y,
             scene.render.resolution_percentage, scene.render.filepath,
             scene.cycles.samples)
    target = Vector((-1.2014189, -8.3727684, 1.365))
    camera.location = target + Vector((0.16, -3.5, 1.15))
    camera.rotation_euler = (target-camera.location).to_track_quat('-Z','Y').to_euler()
    camera.data.ortho_scale = 1.32
    scene.render.resolution_x = 560
    scene.render.resolution_y = 640
    scene.render.resolution_percentage = 100
    scene.cycles.samples = 64
    scene.render.filepath = str(path)
    bpy.context.view_layer.update()
    assert 'FINISHED' in bpy.ops.render.render(write_still=True)
    camera.location, camera.rotation_euler, camera.scale = transform
    camera.data.ortho_scale = state[0]
    (scene.render.resolution_x,scene.render.resolution_y,
     scene.render.resolution_percentage,scene.render.filepath,
     scene.cycles.samples) = state[1:]
    bpy.context.view_layer.update()

if __name__ == '__main__':
    bpy.ops.wm.open_mainfile(filepath=str(ROOT/'native/INPUT_I2.blend'))
    bpy.context.scene.frame_set(1)
    closeup(bpy.context.scene, ROOT/'outputs/GNOME_BEFORE.png')
