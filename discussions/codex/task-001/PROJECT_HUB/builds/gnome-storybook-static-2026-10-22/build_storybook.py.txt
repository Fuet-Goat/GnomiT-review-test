"""Guide-only native storybook form/material pass; no action/world edits."""
from pathlib import Path
import sys,json,math,hashlib
import bpy
from mathutils import Vector,Matrix
ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT/'tools'))
import record_helpers as p
from build_cartoon_guide import descends,action_records,material_records
SOURCE=ROOT/'native/INPUT_K.blend';OUT=ROOT/'native/GNOME_STORYBOOK_L2.blend'
EXPECTED='c09994b1fe6820c537b56b5699ef8099aeded4645c9c667db338cbc731d8a220'
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def write(path,obj):path.write_bytes((json.dumps(obj,indent=2)+'\n').encode())

def painted(source,shadow,light,roughness=.65,blush=False):
    source.use_fake_user=True;mat=source.copy();mat.name=source.name+' — Storybook L'
    tree=mat.node_tree;nodes=tree.nodes;links=tree.links
    nodes.clear();out=nodes.new('ShaderNodeOutputMaterial');bsdf=nodes.new('ShaderNodeBsdfPrincipled')
    bsdf.inputs['Roughness'].default_value=roughness;bsdf.inputs['Specular IOR Level'].default_value=.24
    links.new(bsdf.outputs['BSDF'],out.inputs['Surface']);mat.diffuse_color=p.g.color(light)
    tex=nodes.new('ShaderNodeTexCoord');sep=nodes.new('ShaderNodeSeparateXYZ')
    links.new(tex.outputs['Generated'],sep.inputs[0])
    ramp=nodes.new('ShaderNodeValToRGB');ramp.name='Broad painted colour wash'
    ramp.color_ramp.elements[0].position=.05;ramp.color_ramp.elements[0].color=p.g.color(shadow)
    ramp.color_ramp.elements[1].position=.95;ramp.color_ramp.elements[1].color=p.g.color(light)
    ramp.color_ramp.interpolation='EASE';links.new(sep.outputs['Z'],ramp.inputs[0])
    noise=nodes.new('ShaderNodeTexNoise');noise.name='Quiet broad pigment variation'
    noise.inputs['Scale'].default_value=5;noise.inputs['Detail'].default_value=1
    links.new(tex.outputs['Generated'],noise.inputs['Vector'])
    grain=nodes.new('ShaderNodeValToRGB');grain.color_ramp.elements[0].color=(.94,.94,.94,1)
    grain.color_ramp.elements[1].color=(1,1,1,1);links.new(noise.outputs['Fac'],grain.inputs[0])
    mix=nodes.new('ShaderNodeMixRGB');mix.blend_type='MULTIPLY';mix.inputs[0].default_value=.35
    links.new(ramp.outputs['Color'],mix.inputs[1]);links.new(grain.outputs['Color'],mix.inputs[2])
    color=mix.outputs[0]
    if blush:
        masks=[]
        for x in (.20,.80):
            distance=nodes.new('ShaderNodeVectorMath');distance.operation='DISTANCE';distance.inputs[1].default_value=(x,.04,.44)
            links.new(tex.outputs['Generated'],distance.inputs[0])
            feather=nodes.new('ShaderNodeMapRange');feather.clamp=True
            feather.inputs['From Min'].default_value=.04;feather.inputs['From Max'].default_value=.27
            feather.inputs['To Min'].default_value=.42;feather.inputs['To Max'].default_value=0
            links.new(distance.outputs['Value'],feather.inputs['Value']);masks.append(feather.outputs[0])
        maximum=nodes.new('ShaderNodeMath');maximum.operation='MAXIMUM'
        for i,mask in enumerate(masks):links.new(mask,maximum.inputs[i])
        rosy=nodes.new('ShaderNodeMixRGB');rosy.name='Soft cheek colour, not raised cheek spheres'
        links.new(maximum.outputs[0],rosy.inputs[0]);links.new(color,rosy.inputs[1]);rosy.inputs[2].default_value=p.g.color('EA8E75')
        color=rosy.outputs[0]
    links.new(color,bsdf.inputs['Base Color'])
    mat['native_pipeline_note']='Procedural storybook colour only. Bake/recreate before Godot/glTF integration.'
    return mat

def main():
    assert sha(SOURCE)==EXPECTED and not OUT.exists()
    bpy.context.preferences.filepaths.save_version=0
    bpy.ops.wm.open_mainfile(filepath=str(SOURCE));scene=bpy.context.scene;scene.frame_set(1);bpy.context.view_layer.update()
    root=bpy.data.objects['Guide_Root'];guide=[o for o in scene.objects if descends(o,root)]
    world={o.name:p.object_record(o) for o in scene.objects if o not in guide}
    transforms={o.name:([list(row) for row in o.matrix_basis],o.parent.name if o.parent else None) for o in guide}
    settings=p.g.settings_record(scene);actions=action_records();materials=material_records()
    palette=[('cap soft','B82E26','EB5D3C',.58),('cap folded','B63227','DF5739',.64),
             ('teal cloth','1D6267','419D91',.73),('teal raised','22696B','4CA596',.73),
             ('beard warm','DCCCAE','FFF2D7',.71),('sculpted peach','DB946D','FFD0A1',.60),
             ('warm peach skin','D99A73','F8C295',.61),('leather chestnut','713E2C','9A623D',.68)]
    copies={}
    for o in guide:
        for slot in o.material_slots:
            source=slot.material
            if not source:continue
            for key,shadow,light,rough in palette:
                if key in source.name:
                    if source.name not in copies:copies[source.name]=painted(source,shadow,light,rough,'sculpted peach' in source.name)
                    slot.material=copies[source.name];break
    # A folded band with a broad cloth surface, not a thin tubular rim.
    rim=bpy.data.objects['Guide_cap thick folded rim'];rim.data.use_fake_user=True
    face=bpy.data.objects['Guide_integrated sculpted face'];canonical=face.matrix_world.copy()
    to_local=rim.matrix_world.inverted()@canonical
    rings=[(.880,.204,.151),(.887,.219,.165),(.920,.220,.166),(.936,.205,.152)]
    verts=[];polys=[];count=48
    for z,rx,ry in rings:
        for i in range(count):
            a=math.tau*i/count;verts.append(to_local@Vector((rx*math.cos(a),.003+ry*math.sin(a),z)))
    for k in range(3):
        for i in range(count):
            j=(i+1)%count;a=k*count;b=a+count;polys.append((a+i,a+j,b+j,b+i))
    polys += [tuple(reversed(range(count))),tuple(3*count+i for i in range(count))]
    mesh=bpy.data.meshes.new('Guide_L_folded_cloth_band');mesh.from_pydata(verts,[],polys);mesh.update()
    for m in rim.data.materials:mesh.materials.append(m)
    for f in mesh.polygons:f.use_smooth=True
    rim.data=mesh
    if not any(m.type=='SUBSURF' for m in rim.modifiers):
        sub=rim.modifiers.new('Soft folded cloth L','SUBSURF');sub.levels=2;sub.render_levels=2
    beard=bpy.data.objects['Guide_continuous jaw wrapping beard']
    to_head=canonical.inverted()@beard.matrix_world;back=to_head.inverted()
    for vertex in beard.data.vertices:
        v=to_head@vertex.co
        front=max(0,min(1,(-v.y-.025)/.10));lower=max(0,min(1,(.70-v.z)/.14))
        v.y-=.009*front*lower*(.5+.5*math.cos(5*math.pi*(v.x+.008)/.20))
        v.z+=.010*front*lower*math.cos(3*math.pi*(v.x+.006)/.20)
        v.x*=1+.12*lower
        v.x+=.005*lower*lower
        vertex.co=back@v
    beard.data.update()
    # The first trial's extra shoulder volumes looked like pads. Keep the
    # original sleeves and their parents; do not add a sphere over each joint.
    added=[]
    bpy.context.view_layer.update()
    def validate():
        assert {n:p.object_record(bpy.data.objects[n]) for n in world}==world
        assert {n:([list(row) for row in bpy.data.objects[n].matrix_basis],bpy.data.objects[n].parent.name if bpy.data.objects[n].parent else None) for n in transforms}==transforms
        assert action_records()==actions and p.g.settings_record(scene)==settings
        mats=material_records();assert all(mats[n]==v for n,v in materials.items())
    validate()
    note=bpy.data.texts.new('START_HERE_STORYBOOK_L')
    note.write('Static form/material candidate toward supplied hybrid2.5Dstorybook environment. Same guide identity, copied procedural painted materials, broad folded cap band and broad beard locks. First-trial shoulder pads rejected and absent. Existing actions and world unchanged. New shapes have not had a new animation visual review. Procedural shaders require baking/recreation for Godot. See package receipts.\n')
    assert 'FINISHED' in bpy.ops.wm.save_as_mainfile(filepath=str(OUT),check_existing=False)
    bpy.ops.wm.open_mainfile(filepath=str(OUT));scene=bpy.context.scene;scene.frame_set(1);bpy.context.view_layer.update();validate()
    scene.render.filepath=str(ROOT/'outputs/MAP_STORYBOOK_L.png')
    assert 'FINISHED' in bpy.ops.render.render(write_still=True)
    write(ROOT/'documentation/STORYBOOK_RECEIPT.json',{'status':'NATIVE_SAVED_REOPENED_RENDERED',
        'input_sha256':EXPECTED,'output_sha256':sha(OUT),'reference_sha256':sha(ROOT/'reference/STYLE_REFERENCE.png'),
        'protected_world_objects':len(world),'inherited_guide_transforms_parents_unchanged':True,
        'all_actions_unchanged':True,'original_materials_retained_unchanged':True,
        'added_guide_objects':added,'copied_painted_materials':len(copies),'new_animation_render':'NOT_PERFORMED',
        'Godot_integration_or_shader_bake':'NOT_PERFORMED','human_visual_choice':'PENDING'})
    print('STORYBOOK_NATIVE_COMPLETE',flush=True)

if __name__=='__main__':main()
