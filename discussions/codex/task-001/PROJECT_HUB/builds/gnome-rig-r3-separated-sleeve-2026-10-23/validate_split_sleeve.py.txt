"""Measured split-sleeve evidence; shoulder coverage still needs rendered review."""
import math
from collections import Counter

FRAMES = (1, 37, 55, 73, 91, 109, 145)
BOOTS = ('R1_Boot_-1', 'R1_Boot_1', 'R2 curved sole -1', 'R2 curved sole 1')


def _require(ok, message):
    if not ok:
        raise ValueError('R3 split-sleeve validation: ' + message)


def _distance(a, b):
    return math.sqrt(sum((x-y)**2 for x, y in zip(a, b)))


def _delta(a, b):
    _require(len(a) == len(b), 'Evaluated vertex/face count changed')
    return max((_distance(x, y) for x, y in zip(a, b)), default=0.0)


def _edge_ratios(rest, current, edges):
    ratios = [_distance(current[a], current[b]) / _distance(rest[a], rest[b])
              for a, b in edges if _distance(rest[a], rest[b]) > 1e-10]
    _require(bool(ratios), 'No measurable rest edges')
    return min(ratios), max(ratios)


def _geometry(obj, graph):
    evaluated = obj.evaluated_get(graph)
    mesh = evaluated.to_mesh(preserve_all_data_layers=True, depsgraph=graph)
    try:
        points = [evaluated.matrix_world @ v.co for v in mesh.vertices]
        edges = [tuple(e.vertices) for e in mesh.edges]
        faces = [tuple(p.vertices) for p in mesh.polygons]
        _require(bool(points) and all(math.isfinite(c) for p in points for c in p), obj.name + ' has empty/nonfinite geometry')
        return points, edges, faces
    finally:
        evaluated.to_mesh_clear()


def _closure(geometry):
    points, edges, faces = geometry
    uses = Counter(tuple(sorted((a, b))) for face in faces for a, b in zip(face, face[1:] + face[:1]))
    counts = [uses[tuple(sorted(e))] for e in edges]
    return {'vertices': len(points), 'faces': len(faces), 'boundary_edges': counts.count(1),
            'wire_edges': counts.count(0), 'overfull_edges': sum(n > 2 for n in counts)}


def validate(scene, asset):
    """Return plain JSON data; restore the calling frame even on validation failure."""
    import bpy
    from mathutils import Vector
    objects = {obj.name: obj for obj in asset.all_objects}
    sleeve, patch, coat = [objects[n] for n in ('R3_RightSleeve', 'R3_RightTorsoPatch', 'R2_ContinuousCoat')]
    rig = bpy.data.objects['RigR1_GnomeArmature']; root = bpy.data.objects['R2_GnomeRoot']
    required = {'Root', 'Chest', 'Head', 'upperarm.R', 'forearm.R', 'hand.R', 'upperarm.L', 'forearm.L', 'hand.L', 'lantern'}
    _require(set(rig.data.bones.keys()) == required, 'Expected the original ten bones')
    _require(sleeve.data != patch.data and sleeve.data != coat.data, 'Sleeve must have independent mesh topology')
    mods = list(sleeve.modifiers); arms = [m for m in mods if m.type == 'ARMATURE']
    _require(len(arms) == 1 and arms[0].object == rig and arms[0].use_deform_preserve_volume, 'Sleeve requires original rig and Preserve Volume')
    _require(any(m.type == 'SUBSURF' and m.show_viewport and m.show_render for m in mods[:mods.index(arms[0])]), 'Subdivision must precede Armature')
    weight_report = {}
    for obj, allowed in ((sleeve, {'upperarm.R', 'forearm.R'}), (patch, {'Root', 'Chest'})):
        names = {g.index: g.name for g in obj.vertex_groups}; values = []
        binding = [m for m in obj.modifiers if m.type == 'ARMATURE']
        _require(len(binding) == 1 and binding[0].object == rig and binding[0].show_viewport and binding[0].show_render and not binding[0].use_bone_envelopes, obj.name + ' has an invalid rig binding')
        for vertex in obj.data.vertices:
            _require(all(math.isfinite(g.weight) and 0 <= g.weight <= 1 for g in vertex.groups), obj.name + ' has nonfinite/out-of-range weights')
            weights = {names[g.group]: float(g.weight) for g in vertex.groups if g.weight > 1e-8}
            _require(set(weights) <= allowed and abs(sum(weights.values())-1) < 1e-5, obj.name + ' has invalid bone weights')
            values.append(weights.get('forearm.R', 0.0))
        weight_report[obj.name] = {'allowed_bones': sorted(allowed), 'vertices': len(values)}
        if obj == sleeve:
            _require(len(values) == 37*24, 'Expected original R2 sampled sleeve rings')
            centres = [sum((v.co for v in obj.data.vertices[i:i+24]), Vector())/24 for i in range(0, len(values), 24)]
            arc = 0.0; previous = 0.0; jumps = []
            for i, centre in enumerate(centres):
                if i: arc += (centre-centres[i-1]).length
                ring = values[i*24:(i+1)*24]; weight = sum(ring)/24
                _require(max(ring)-min(ring) < 1e-5 and weight >= previous-1e-5, 'Sleeve weights vary around a ring or reverse along the sleeve')
                _require((arc > .17 or weight < 1e-5) and (arc < .29 or weight > 1-1e-5), 'Forearm blend extends beyond authored .17-.29 arc band')
                jumps.append(weight-previous); previous = weight
            weight_report[obj.name].update({'centreline_length_local': arc, 'maximum_adjacent_ring_weight_step': max(jumps)})
    original_frame, original_subframe = scene.frame_current, scene.frame_subframe
    try:
        scene.frame_set(1); graph = bpy.context.evaluated_depsgraph_get()
        meshes = {n: o for n, o in objects.items() if o.type == 'MESH'}
        baseline = {n: _geometry(o, graph) for n, o in meshes.items()}
        closure = {n: _closure(g) for n, g in baseline.items()}
        _require(all(closure[sleeve.name][k] == 0 for k in ('boundary_edges', 'wire_edges', 'overfull_edges')), 'Evaluated sleeve is not closed')
        for name in BOOTS: _require(name in baseline, 'Missing planted foot: ' + name)
        cuff = Vector((-.350, -.100, .680)); tangent = Vector((-.021, -.027, .012)).normalized()
        canonical = root.matrix_world.inverted()
        cut_centres = [sum((canonical @ baseline[coat.name][0][i] for i in face), Vector())/len(face) for face in baseline[coat.name][2]]
        retained_cut_faces = sum(c.x < -.05 and c.z > .54 for c in cut_centres)
        cuff_indices = [i for i, p in enumerate(baseline[sleeve.name][0]) if abs((canonical @ p-cuff).dot(tangent)) < .008 and (canonical @ p-cuff).length < .085]
        _require(len(cuff_indices) >= 12, 'Cannot identify evaluated cuff surface near authored cuff plane')
        cuff0 = sum((baseline[sleeve.name][0][i] for i in cuff_indices), Vector())/len(cuff_indices)
        root_pose = rig.pose.bones['Root'].matrix.copy(); root_world = root.matrix_world.copy()
        scale = max(root_world.to_scale()); samples = []; foot_max = patch_max = cuff_error_max = root_max = 0.0
        for frame in FRAMES:
            scene.frame_set(frame); graph = bpy.context.evaluated_depsgraph_get()
            geometry = {n: _geometry(meshes[n], graph) for n in (*BOOTS, sleeve.name, patch.name)}
            foot_delta = max(_delta(baseline[n][0], geometry[n][0]) for n in BOOTS); foot_max = max(foot_max, foot_delta)
            _require(foot_delta <= 1e-5, 'Evaluated feet moved at frame ' + str(frame))
            current = geometry[sleeve.name]
            _require(current[1:] == baseline[sleeve.name][1:], 'Evaluated sleeve connectivity changed')
            edge_min, edge_max = _edge_ratios(baseline[sleeve.name][0], current[0], current[1])
            centre = sum((current[0][i] for i in cuff_indices), Vector())/len(cuff_indices)
            bone = rig.pose.bones['forearm.R']; skin = rig.matrix_world @ bone.matrix @ bone.bone.matrix_local.inverted() @ rig.matrix_world.inverted()
            cuff_error = (centre-skin @ cuff0).length; cuff_error_max = max(cuff_error_max, cuff_error)
            patch_delta = _delta(baseline[patch.name][0], geometry[patch.name][0]); patch_max = max(patch_max, patch_delta)
            faces = baseline[patch.name][2]
            face_delta = max(((sum((geometry[patch.name][0][i]-baseline[patch.name][0][i] for i in face), Vector())/len(face)).length for face in faces), default=0.0)
            root_delta = max(abs(a-b) for old, new in ((root_pose, rig.pose.bones['Root'].matrix), (root_world, root.matrix_world)) for row_a, row_b in zip(old, new) for a, b in zip(row_a, row_b)); root_max = max(root_max, root_delta)
            samples.append({'frame': frame, 'feet_max_vertex_delta_world': foot_delta, 'sleeve_edge_ratio_min': edge_min, 'sleeve_edge_ratio_max': edge_max, 'evaluated_cuff_surface_centroid_world': list(centre), 'cuff_forearm_tracking_error_world': cuff_error, 'torso_patch_max_vertex_delta_world': patch_delta, 'torso_patch_max_face_centroid_delta_world': face_delta})
        _require(root_max <= 1e-6, 'Root driver moved'); _require(cuff_error_max <= 1e-5, 'Cuff does not follow forearm rigidly')
        _require(patch_max <= .02*scale, 'Torso patch motion exceeds small breathing envelope')
        return {'status': 'NUMERICAL_CHECKS_PASSED_VISUAL_REVIEW_PENDING', 'samples': samples, 'evaluated_mesh_closure': closure,
                'weights': weight_report, 'feet_max_vertex_delta_world': foot_max, 'root_driver_max_matrix_delta': root_max,
                'separate_sleeve_topology': True, 'evaluated_cuff_surface_vertex_count': len(cuff_indices),
                'old_coat_evaluated_faces_remaining_in_cut_region': retained_cut_faces,
                'source_bone_and_action_preservation': 'CHECK_SEPARATELY_AGAINST_BUILD_BASELINE',
                'shoulder_cap_coverage': 'VISUAL_UNKNOWN_FRONT_THREE_QUARTER_SIDE_REQUIRED', 'underarm_clearance': 'VISUAL_UNKNOWN_NO_BOUNDING_BOX_VERDICT'}
    finally:
        scene.frame_set(original_frame, subframe=original_subframe)
