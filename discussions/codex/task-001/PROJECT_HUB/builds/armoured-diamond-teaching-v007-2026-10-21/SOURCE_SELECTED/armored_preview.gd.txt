extends Control
const Model=preload("res://scripts/armored_demo_model.gd")
const Diagram=preload("res://scripts/casing_diagram.gd")
const FONT=preload("res://assets/font.ttf")
var model=Model.new()
var page := 0
var event_serial := 0
var title: Label
var explanation: Label
var feedback: Label
var diagram
var hit_button: Button
var invalid_button: Button
var back: Button
var next: Button
var count: Label
var reduced: CheckButton

func _ready() -> void:
	if DisplayServer.get_name()!="headless" and DisplayServer.get_screen_count()>1:DisplayServer.window_set_current_screen(1)
	var weight := FontVariation.new();weight.base_font=FONT
	weight.variation_opentype={TextServerManager.get_primary_interface().name_to_tag("wght"):650.0}
	var style := Theme.new();style.default_font=weight;style.default_font_size=18;theme=style
	var margin := MarginContainer.new();margin.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	for edge in ["left","right","top","bottom"]:margin.add_theme_constant_override("margin_"+edge,22)
	add_child(margin)
	var column := VBoxContainer.new();column.add_theme_constant_override("separation",12);margin.add_child(column)
	_label(column,"GNOMIT · OBJECT GUIDE",17,Color("e7c487"))
	_label(column,"PROPOSED RULE · TEACHING PREVIEW",12,Color("8bb8bc"))
	title=_label(column,"",30)
	explanation=_label(column,"",18);explanation.custom_minimum_size.y=116
	diagram=Diagram.new();diagram.size_flags_vertical=Control.SIZE_EXPAND_FILL;diagram.custom_minimum_size.y=260;column.add_child(diagram)
	feedback=_label(column,"",17,Color("ffdfa3"));feedback.custom_minimum_size.y=72
	hit_button=_button("",try_valid);column.add_child(hit_button)
	invalid_button=_button("Try a distant match",try_invalid);column.add_child(invalid_button)
	var navigation := HBoxContainer.new();navigation.add_theme_constant_override("separation",12);column.add_child(navigation)
	back=_button("‹",func()->void:set_page(page-1));back.custom_minimum_size.x=64;navigation.add_child(back)
	count=_label(navigation,"",17);count.size_flags_horizontal=Control.SIZE_EXPAND_FILL
	next=_button("›",func()->void:set_page(page+1));next.custom_minimum_size.x=64;navigation.add_child(next)
	var bottom := HBoxContainer.new();bottom.add_theme_constant_override("separation",8);column.add_child(bottom)
	var reset_button := _button("Reset",reset_demo);bottom.add_child(reset_button)
	reduced=CheckButton.new();reduced.text="Reduced motion";bottom.add_child(reduced)
	reduced.toggled.connect(func(value:bool)->void:diagram.reduced_motion=value;diagram.transition=1.0;diagram.queue_redraw())
	_label(column,"Illustration only · campaign and save unchanged",12,Color("8bb8bc"))
	reset_demo();set_page(0)
	print("ARMOURED_TEACHING_READY")

func _label(parent: Node, text: String, font_size: int, color: Color=Color("f6edda")) -> Label:
	var label := Label.new();label.text=text;label.autowrap_mode=TextServer.AUTOWRAP_WORD_SMART
	label.add_theme_font_size_override("font_size",font_size);label.add_theme_color_override("font_color",color)
	label.mouse_filter=Control.MOUSE_FILTER_IGNORE;parent.add_child(label);return label

func _button(text: String, action: Callable) -> Button:
	var button := Button.new();button.text=text;button.custom_minimum_size.y=48
	for state in ["normal","hover","pressed","disabled","focus"]:
		var style := StyleBoxFlat.new();style.bg_color=Color("345851") if state!="disabled" else Color("1c3639")
		if state=="hover":style.bg_color=Color("4d7663")
		style.border_color=Color("c3a165") if state!="disabled" else Color("466262")
		style.set_border_width_all(2);style.set_corner_radius_all(12)
		button.add_theme_stylebox_override(state,style)
	button.pressed.connect(action);return button

func set_page(value: int) -> void:
	page=clampi(value,0,1);diagram.page=page;diagram.queue_redraw()
	title.text="Meet stone armour" if page==0 else "Try breaking it"
	explanation.text="A stone casing protects this diamond. The blue window shows its mineral. While protected, it cannot move or join a match." if page==0 else "Match beside the casing twice. Each separate match removes one layer. Then match the revealed diamond to collect it."
	back.disabled=page==0;next.disabled=page==1;count.text="PAGE %d / 2"%(page+1)
	hit_button.visible=page==1;invalid_button.visible=page==1
	if page==0:feedback.text="Stone casing → two hits → ordinary diamond.\nThis is a proposed first variant."
	else:_feedback_for_state()
	_refresh_buttons()

func _next_id() -> String:
	event_serial+=1;return "%d:%d"%[model.generation,event_serial]

func try_valid() -> void:
	if diagram.transition<1.0:return
	var result: Dictionary
	if model.hp>0:
		result=model.resolve(_next_id(),"match_wave",[Vector2i(0,0),Vector2i(0,1),Vector2i(0,2)])
	else:
		result=model.resolve(_next_id(),"mineral_match",[Vector2i(0,1),Vector2i(1,1),Vector2i(2,1)])
	diagram.present(model.snapshot(),result);_feedback_for_state();_refresh_buttons()

func try_invalid() -> void:
	if diagram.transition<1.0:return
	model.resolve(_next_id(),"match_wave",[Vector2i(4,0),Vector2i(4,1),Vector2i(4,2)])
	feedback.text="No damage: that match is too far away.\nOnly an orthogonal neighbour hits the casing."

func reset_demo() -> void:
	model.reset();event_serial=0;diagram.present(model.snapshot());_feedback_for_state();_refresh_buttons()

func _feedback_for_state() -> void:
	if page==0:feedback.text="Stone casing → two hits → ordinary diamond.\nThis is a proposed first variant.";return
	if model.collected:feedback.text="Collected once. Protection removal did not collect it earlier. Reset to try again."
	elif model.hp==2:feedback.text="Two layers remain. Tap below to demonstrate a neighbouring triple."
	elif model.hp==1:feedback.text="2 → 1. The casing cracks; the diamond stays protected."
	else:feedback.text="1 → 0. Armour gone; the diamond remains. Now match it normally."

func _refresh_buttons() -> void:
	hit_button.text="Make a neighbouring match" if model.hp>0 else "Match the diamond"
	hit_button.disabled=model.collected or diagram.transition<1.0
	invalid_button.disabled=model.collected or model.hp==0 or diagram.transition<1.0

func _process(_delta: float) -> void:
	if is_instance_valid(hit_button):_refresh_buttons()

func _input(event: InputEvent) -> void:
	if event is InputEventKey and event.pressed and not event.echo:
		if event.keycode==KEY_LEFT:set_page(page-1)
		elif event.keycode==KEY_RIGHT:set_page(page+1)
