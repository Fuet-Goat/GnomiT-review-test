+	check(bool(frozen_result.get("valid", false)), "Actual native-input Core transaction is valid")
+	check(String(frozen_event.get("power", "")) == scenario, "Frozen Core event belongs to the selected Tornado scenario")
+	check(int(frozen_event.get("origin", -1)) == int(fixture_pair()[1]), "Actual event origin is the fixture's swap destination")
+	check(same_json(active_targets, frozen_event.get("targets", [])), "Signal targets equal actual Core event targets")
+	expected_targets = derive_radial_targets(frozen_event)
+	check(same_json(active_targets, expected_targets), "Actual target list and cardinality equal eligible surrounding 3x3 cells")
+	check(not active_targets.has(int(frozen_event.get("origin", -1))) and active_targets.size() <= 8, "Surrounding targets exclude the origin and fit the eight-cell neighborhood")
+	var expected_removed := expected_targets.duplicate()
+	expected_removed.append(int(frozen_event.get("origin", -1)))
+	expected_removed.sort()
+	check(same_json(frozen_event.get("indices", []), expected_removed), "Actual removal indices contain only origin plus real radial targets")
+	check(same_json(app.board._power_event, frozen_event) and is_zero_approx(float(app.board._power_t)), "Event and state freeze synchronously before the effect advances or redraws")
+	check(int(frozen_state.moves_left) == int(initial_state.moves_left) - 1, "Native activation spends exactly one move")
+	write_json("EVENT", frozen_event)
+	write_json("RESULT", frozen_result)
+	write_json("ACTIVATION_STATE", frozen_state)
+	observations.append({
+		"phase": "event_frozen_before_draw", "process_frame": relative_frame(),
+		"actual_targets": active_targets.duplicate(), "actual_target_count": active_targets.size(),
+		"derived_eligible_targets": expected_targets.duplicate(),
+	})

+func shot(phase: String, requested_progress: float = -1.0) -> void:
+	await RenderingServer.frame_post_draw
+	var path := output + "/" + scenario + "_" + phase + ".png"
+	var captured := root.get_texture().get_image()
+	check(captured.get_size() == SCREEN_SIZE, "Entire root capture is exactly 432x960")
+	check(captured.save_png(path) == OK, "Screenshot saves " + path)
+	observations.append({
+		"phase": phase, "file": path, "process_frame": relative_frame(),
+		"requested_progress": requested_progress, "actual_power_t": app.board._power_t,
+		"power_active": not app.board._power_event.is_empty(),
+		"actual_targets": active_targets.duplicate(), "actual_target_count": active_targets.size(),
+		"moves_left": app.core.moves_left,
+	})

+func wait_for_phase(progress: float) -> bool:
+	while capture_time_available():
+		if activation_frame >= 0:
+			if app.board._power_event.is_empty():
+				check(false, "Effect ended before requested phase " + str(progress))
+				return false
+			if float(app.board._power_t) >= progress:
+				return true
+		await next_frame()
+	check(false, "Seven-second capture ended before phase " + str(progress))
+	return false
+
+func validate_fixture() -> void:
+	check(app.scenario == scenario and app.seed_value == SEED, "Actual lab is reset to the selected seeded scenario")
+	check(app.core.fixture_pair == fixture_pair(), "Existing fixture pair is unchanged")
+	check(app.core.cols == 7 and app.core.rows == 8, "Existing seven-column eight-row fixture dimensions are unchanged")
+	check(app.core.grid.size() == 56 and app.core.floor.size() == 56 and app.core.runes.size() == 56, "Complete fixture arrays are present")
+	if app.core.grid.size() != 56 or app.core.floor.size() != 56 or app.core.runes.size() != 56:
+		return
+	var special_kind: int = app.core.TORNADO_V if scenario == "tornado_v" else app.core.TORNADO_H
+	check(int(app.core.grid[int(fixture_pair()[0])]) == special_kind, "Real fixture starts with the selected special at its established source")
+	check(int(app.core.grid[int(fixture_pair()[1])]) >= 0 and int(app.core.grid[int(fixture_pair()[1])]) < app.core.GEM_KINDS, "Real fixture swap destination is an ordinary mineral")
+	check(int(app.core.grid[24]) == -2 and int(app.core.grid[30]) == -2, "Established gap cells remain unchanged")
+	check(int(app.core.grid[39]) == -3 and int(app.core.runes[39]) == 1, "Established rune remains unchanged")
+	check(int(app.core.floor[23]) == 1 and int(app.core.floor[26]) == 1 and int(app.core.floor[43]) == 1, "Established soil fixtures remain unchanged")
+	check(app.core.moves_left == 12 and app.last_result.is_empty(), "Capture starts before any activation with the original move budget")
+
+func validate_geometry() -> void:
+	var board_rect: Rect2 = app.board.get_global_rect()
+	var cells_rect: Rect2 = app.board.get_global_transform() * app.board._board_rect()
+	var viewport_rect := Rect2(Vector2.ZERO, Vector2(SCREEN_SIZE))
+	var actual_window := DisplayServer.window_get_size()
+	check(root.size == SCREEN_SIZE and root.content_scale_size == SCREEN_SIZE, "Root and content viewport remain 432x960")
+	check(actual_window == SCREEN_SIZE, "Actual native window content size is 432x960")
+	check(DisplayServer.window_get_current_screen() == 0, "Actual native window is on monitor 0")
+	check(viewport_rect.encloses(board_rect) and viewport_rect.encloses(cells_rect), "Entire board control and all cells fit inside the portrait capture")
+	check(board_rect.size.x >= 300.0 and board_rect.size.y >= 350.0, "Actual board meets the established usable minimum size")
+	geometry = {
+		"native_screen": DisplayServer.window_get_current_screen(),
+		"native_window_size": [actual_window.x, actual_window.y],
+		"root_size": [root.size.x, root.size.y],
+		"content_scale_size": [root.content_scale_size.x, root.content_scale_size.y],
+		"board_rect": [board_rect.position.x, board_rect.position.y, board_rect.size.x, board_rect.size.y],
+		"cells_rect": [cells_rect.position.x, cells_rect.position.y, cells_rect.size.x, cells_rect.size.y],
+	}

+func run_capture() -> void:
+	root.content_scale_size = SCREEN_SIZE
+	root.size = SCREEN_SIZE
+	if DisplayServer.get_name() == "headless":
+		check(false, "Native Tornado capture requires a graphical Godot driver")
+		finish_capture()
+		return
+	DisplayServer.window_set_current_screen(0)
+	var lab_script = load(LAB_PATH)
+	if lab_script == null or not lab_script.can_instantiate():
+		check(false, "Actual natural power lab script must load")
+		finish_capture()
+		return
+	app = lab_script.new()
+	root.add_child(app)
+	app.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
+	await wait_frames(3)
+	check(app.board != null and app.board.get_script().resource_path == BOARD_PATH, "Actual lab uses the isolated magic_volcano_board renderer")
+	if not failures.is_empty():
+		finish_capture()
+		return
+	app.reduced.button_pressed = false
+	app.seed_value = SEED
+	app.reset_scenario(scenario)
+	app.board.power_started.connect(on_power_started)
+	await wait_frames(2)
+	validate_fixture()
+	validate_geometry()
+	initial_state = core_state(app.core)
+	write_json("INITIAL_STATE", initial_state)
+	if not failures.is_empty():
+		finish_capture()
+		return
+	while relative_frame() < CAPTURE_FPS:
+		await next_frame()
+	await shot("00_ready")
+	await native_fixture_drag()
+	for phase in [["01_start", 0.0], ["02_telegraph", 0.12], ["03_swirl", 0.45], ["04_clear", 0.85]]:
+		if await wait_for_phase(float(phase[1])):
+			await shot(String(phase[0]), float(phase[1]))
+		else:
+			break
+	while (app._resolving or app.board.busy) and capture_time_available():
+		await next_frame()
+	check(not app._resolving and not app.board.busy, "Actual lab finishes its complete result inside the seven-second capture")
+	check(not frozen_event.is_empty(), "Actual Core event froze before effect rendering")
+	check(app.core.moves_left == int(initial_state.moves_left) - 1, "Completed native gesture spends exactly one move")
+	check(same_json(core_state(app.core), frozen_state), "Complete Core/RNG state remains equal to the synchronous resolution snapshot")
+	check(display_matches(), "Final displayed grid, soil and runes match actual resolved Core")
+	check(presentation_clear(), "Final presentation has no busy stream, hidden cells, moving stones or stale power overlay")
+	write_json("FINAL_STATE", core_state(app.core))
+	if capture_time_available():
+		await shot("05_final")
+	while capture_time_available():
+		await next_frame()
+	check(relative_frame() == CAPTURE_FRAMES, "Capture contains exactly 210 process frames at fixed 30 fps")
+	check(presentation_clear() and display_matches(), "Final hold remains clean and equal to the resolved Core")
+	check(invariant_sample_count > 0 and not invariant_failed and not result_failed, "Every sampled frame after activation preserves complete Core/RNG and actual result")
+	check(effect_end_frame >= activation_frame and activation_frame >= 0, "Actual effect start and cleanup frames are measured")
+	check(absf(effect_sampled_seconds - EXPECTED_EFFECT_SECONDS) <= TIMING_TOLERANCE_SECONDS, "Measured active effect duration agrees with 1.10 seconds within two frame intervals")
+	finish_capture()
+
+func finish_capture() -> void:
+	if ended:
+		return
+	ended = true
+	var effect_frames := effect_end_frame - activation_frame if effect_end_frame >= 0 and activation_frame >= 0 else -1
+	var receipt := {
+		"status": "PASS" if failures.is_empty() else "FAIL", "failures": failures,
+		"scenario": scenario, "seed": SEED, "fixture_pair": fixture_pair(),
+		"actual_targets": active_targets, "actual_target_count": active_targets.size(),
+		"derived_eligible_targets": expected_targets, "derived_target_count": expected_targets.size(),
+		"geometry": geometry, "fixed_fps_expected": CAPTURE_FPS,
+		"clip_process_frames": relative_frame(), "clip_seconds_at_fixed_fps": float(relative_frame()) / float(CAPTURE_FPS),
+		"expected_effect_seconds": EXPECTED_EFFECT_SECONDS,
+		"effect_start_frame": activation_frame - start_frame if activation_frame >= 0 else -1,
+		"effect_end_frame": effect_end_frame - start_frame if effect_end_frame >= 0 else -1,
+		"effect_elapsed_process_frames": effect_frames,
+		"effect_frame_seconds_at_fixed_fps": float(effect_frames) / float(CAPTURE_FPS) if effect_frames >= 0 else -1.0,
+		"effect_sampled_delta_seconds": effect_sampled_seconds,
+		"effect_delta_from_1_10_seconds": effect_sampled_seconds - EXPECTED_EFFECT_SECONDS if activation_frame >= 0 else null,
+		"effect_timing_tolerance_seconds": TIMING_TOLERANCE_SECONDS,
+		"sampled_frame_delta_seconds": sampled_frame_delta_seconds,
+		"ignored_uninitialized_startup_delta_frames": ignored_startup_delta_frames,
+		"resolved_core_rng_invariant_sample_count": invariant_sample_count,
+		"observations": observations, "frame_observations": frame_observations,
+		"native_input_events": gesture_events,
+		"proof": "Actual Godot lab root renders and Input.parse_input_event press/drag/release; synchronous actual Core event/result/state freeze; no human, Android or visual acceptance claim",
+	}
+	write_json("REPORT", receipt)
+	print("TORNADO_TOP_CAPTURE " + JSON.stringify({"status": receipt.status, "scenario": scenario, "failures": failures, "actual_targets": active_targets, "clip_process_frames": relative_frame()}))
+	quit(0 if failures.is_empty() else 1)

+func run() -> void:
+	start_frame = Engine.get_process_frames()
+	output = ProjectSettings.globalize_path(OUTPUT_PATH)
+	check(DirAccess.make_dir_recursive_absolute(output) == OK, "Dedicated Tornado output directory exists")
+	var arguments := OS.get_cmdline_user_args()
+	var valid_arguments := arguments.is_empty() or (arguments.size() == 1 and String(arguments[0]) in ["--tornado-h", "--tornado-v"])
+	if not valid_arguments:
+		check(false, "Use no user argument for tornado_h, or exactly one of --tornado-h / --tornado-v")
+		finish_capture()
+		return
+	scenario = "tornado_v" if "--tornado-v" in arguments else "tornado_h"
+	await run_capture()
