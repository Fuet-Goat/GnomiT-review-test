def ground(x,y):
    return .14+.045*math.sin(x*.75)*math.cos(y*.65)+.026*math.cos(y*1.3)

def ring_mesh(name,rings,materials):
    n=len(rings[0]);vs=[p for r in rings for p in r];fs=[]
    for j in range(len(rings)-1):
        for i in range(n):fs.append((j*n+i,j*n+(i+1)%n,(j+1)*n+(i+1)%n,(j+1)*n+i))
    fs.append(tuple(range(n-1,-1,-1)))
    fs.append(tuple((len(rings)-1)*n+i for i in range(n)))
    me=bpy.data.meshes.new(name);me.from_pydata(vs,[],fs);me.update()
    ob=bpy.data.objects.new(name,me);bpy.context.collection.objects.link(ob)
    finish(ob,name,materials[0],.035,True)
    for m in materials[1:]:me.materials.append(m)
    return ob

# Irregular round-sided island with a shallow turf lip and warm soil cut.
n=128
def outline(i):
    a=i/n*math.tau;w=1+.022*math.sin(a*5)+.016*math.cos(a*9)
    return 4.6*math.copysign(abs(math.cos(a))**.78,math.cos(a))*w,5.5*math.copysign(abs(math.sin(a))**.8,math.sin(a))*w
edge=[outline(i) for i in range(n)]

def rock(name,x,y,s):
    z=ground(x,y)
    ob=ball(name,(x,y,z+s*.19),(s*.72,s*.56,s*.52),rockmats[rng.randrange(3)],3)
    for v in ob.data.vertices:
        a=math.atan2(v.co.y,v.co.x);v.co*=1+.065*math.sin(a*3+v.co.z*4)+.025*math.cos(a*7)
    ob.rotation_euler.z=rng.random()*math.tau
    cap=ball(name+' seated moss',(x-.05*s,y+.025*s,z+s*.54),(s*.46,s*.38,s*.13),moss,2)
    return ob
