from pathlib import Path
import bpy,json,math
ROOT=Path(__file__).resolve().parents[1]
def shoulder_metrics(path,end):
    bpy.ops.wm.open_mainfile(filepath=str(path));scene=bpy.context.scene
    angles=[]
    for f in range(1,end+1):
        scene.frame_set(f);angles.append(math.degrees(bpy.data.objects['Guide_ShoulderWave'].rotation_euler.y))
    return {'excursion_degrees':max(angles)-min(angles),'max_degrees_per_frame':max(abs(b-a) for a,b in zip(angles,angles[1:]))}
old=shoulder_metrics(ROOT/'native/INPUT_J.blend',145)
new=shoulder_metrics(ROOT/'native/GNOME_SOFT_GREETING_K.blend',121)
bpy.ops.wm.open_mainfile(filepath=str(ROOT/'native/GNOME_K_ANIMATION_STAGE.blend'))
scene=bpy.context.scene;assert scene.render.fps==24
nodes=[o for o in scene.objects if o.animation_data and o.animation_data.action]
assert len(nodes)==9
scene.frame_set(1);bpy.context.view_layer.update()
first={o.name:[v for row in o.matrix_basis for v in row] for o in nodes}
scene.frame_set(121);bpy.context.view_layer.update()
last={o.name:[v for row in o.matrix_basis for v in row] for o in nodes}
error=max(abs(x-y) for n in first for x,y in zip(first[n],last[n]));assert error<1e-6,error
assert new['excursion_degrees']<old['excursion_degrees'] and new['max_degrees_per_frame']<old['max_degrees_per_frame']
record={'stage_fresh_open':'PASS','stage_objects':len(scene.objects),'active_animated_nodes':len(nodes),
        'first_last_pose_max_basis_error':error,'old_shoulder':old,'new_shoulder':new,
        'interpretation':'Measures excursion and maximum step only; does not prove human-rated fluidity.'}
(ROOT/'documentation/MOTION_STAGE_CHECK.json').write_bytes((json.dumps(record,indent=2)+'\n').encode())
print(json.dumps(record))
