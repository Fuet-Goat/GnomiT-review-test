+extends Node
+## Campaign presentation adapter. The board owns targets and the event clock;
+## this node owns exactly one ordinary stone voice and the established power banks.
+const RockAudio = preload("res://scripts/rock_audio.gd")
+const MegaAudio = preload("res://scripts/lab/mega_volcano_v4_audio.gd")
+const VolcanoAudio = preload("res://scripts/lab/volcano_audio.gd")
+const TornadoAudio = preload("res://scripts/lab/tornado_audio.gd")
+const LightningAudio = preload("res://scripts/lab/lightning_audio.gd")
+const ElementAudio = preload("res://scripts/lab/reward_rules_audio.gd")
+
+var stone_audio
+var mega_audio
+var volcano_audio
+var tornado_audio
+var storm_audio
+var element_audio
+var enabled := true
+var _board: Node
+var _attachment_id := 0
+var _power_callback: Callable
+var _birth_callback: Callable
+var _exit_callback: Callable
+
+func _ready() -> void:
+	stone_audio = RockAudio.new()
+	mega_audio = MegaAudio.new()
+	volcano_audio = VolcanoAudio.new()
+	tornado_audio = TornadoAudio.new()
+	storm_audio = LightningAudio.new()
+	element_audio = ElementAudio.new()
+	for bank in _banks(): add_child(bank)
+	stone_audio.set_enabled(enabled)
+
+func attach_board(board: Node) -> void:
+	# Old signal senders and their sounds lose ownership before the new board
+	# is attached. In particular an old tree_exiting cannot stop a new eruption.
+	_disconnect_board()
+	stop_all()
+	_attachment_id += 1
+	_board = board
+	if not is_instance_valid(_board): return
+	_power_callback = _on_power_started.bind(_board, _attachment_id)
+	_birth_callback = _on_reward_born.bind(_board, _attachment_id)
+	_exit_callback = _on_board_exiting.bind(_board, _attachment_id)
+	_board.connect("power_started", _power_callback)
+	_board.connect("reward_born", _birth_callback)
+	_board.tree_exiting.connect(_exit_callback)
+
+func set_enabled(value: bool) -> void:
+	enabled = value
+	if is_instance_valid(stone_audio): stone_audio.set_enabled(value)
+	if not value: stop_all()
+
+func play_foley(kind: String, strength: float = 1.0) -> bool:
+	# Main delegates its existing board callback here; no second connection
+	# or RockAudio instance is made, including on ordinary cascade clears.
+	if kind not in ["move", "break"]: return false
+	if not enabled or not is_instance_valid(stone_audio): return true
+	if kind == "break":
+		var bank = _current_power_bank()
+		if is_instance_valid(bank) and bank.event_active: return true
+	stone_audio.play_cue(kind, strength)
+	return true
+
+func stop_all() -> void:
+	for bank in _banks():
+		if is_instance_valid(bank): bank.stop_all()
+
+func _banks() -> Array:
+	return [stone_audio, mega_audio, volcano_audio, tornado_audio, storm_audio, element_audio]
+
+func _owns(source: Node, attachment: int) -> bool:
+	return is_instance_valid(source) and source == _board and attachment == _attachment_id
+
+func _on_power_started(power: String, targets: Array, source: Node, attachment: int) -> void:
+	if not enabled or not _owns(source, attachment) or _active_power() != power: return
+	var reduced: bool = _board.reduced_motion
+	match power:
+		"mega_volcano": mega_audio.begin_mega(targets, reduced)
+		"volcano":
+			if _board.painted_volcano: volcano_audio.begin(targets, reduced)
+		"tornado_h", "tornado_v": tornado_audio.begin(reduced)
+		"lightning_storm": storm_audio.begin(reduced)
+		"black_hole":
+			var kinds: Array = []
+			for source_power in _board.fusion_sources(): kinds.append(int(source_power.kind))
+			element_audio.begin_fusion(kinds, reduced)
+		"earthquake": element_audio.begin(power, reduced)
+
+func _on_reward_born(awards: Array, source: Node, attachment: int) -> void:
+	if not enabled or not _owns(source, attachment) or awards.is_empty(): return
+	# The board emits one birth event for the entire award group.
+	element_audio.begin_birth(int(awards[0].get("birth_tier", 5)), _board.reduced_motion)
+
+func _active_power() -> String:
+	if not is_instance_valid(_board) or not _board.is_inside_tree(): return ""
+	if _board._power_event.is_empty() or _board._power_epoch != _board._epoch: return ""
+	return String(_board._power_event.get("power", ""))
+
+func _current_power_bank():
+	match _active_power():
+		"mega_volcano": return mega_audio
+		"volcano": return volcano_audio if _board.painted_volcano else null
+		"tornado_h", "tornado_v": return tornado_audio
+		"lightning_storm": return storm_audio
+		"black_hole", "earthquake": return element_audio
+	return null
+
+func _process(_delta: float) -> void:
+	if not enabled: return
+	var active := _active_power()
+	if is_instance_valid(mega_audio) and mega_audio.event_active:
+		if active == "mega_volcano": mega_audio.sync_progress(_board._power_t)
+		else: mega_audio.finish_event()
+	if is_instance_valid(volcano_audio) and volcano_audio.event_active:
+		if active == "volcano" and _board.painted_volcano: volcano_audio.sync_progress(_board._power_t)
+		else: volcano_audio.finish_event()
+	if is_instance_valid(tornado_audio) and tornado_audio.event_active and active not in ["tornado_h", "tornado_v"]:
+		tornado_audio.finish_event()
+	if is_instance_valid(storm_audio) and storm_audio.event_active and active != "lightning_storm":
+		storm_audio.finish_event()
+	if is_instance_valid(element_audio) and element_audio.event_active and active not in ["black_hole", "earthquake"]:
+		element_audio.finish_event()
+
+func _on_board_exiting(source: Node, attachment: int) -> void:
+	if not _owns(source, attachment): return
+	attach_board(null)
+
+func _disconnect_board() -> void:
+	if not is_instance_valid(_board): return
+	for binding in [["power_started", _power_callback], ["reward_born", _birth_callback], ["tree_exiting", _exit_callback]]:
+		if _board.is_connected(binding[0], binding[1]): _board.disconnect(binding[0], binding[1])
+
+func _exit_tree() -> void:
+	_disconnect_board()
+	_board = null
+	stop_all()
