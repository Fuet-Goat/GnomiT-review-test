+func _apply_atomic_power(power: String, sources: Array, components: Array, events: Array, extra: Dictionary = {}) -> void:
+	if power != "power_combo" or sources.size() != 2 or components.size() != 2:
+		super._apply_atomic_power(power, sources, components, events, extra)
+		return
+	for source in sources:
+		if int(source.get("kind", -1)) != VOLCANO or int(source.get("birth_tier", -1)) != 4:
+			super._apply_atomic_power(power, sources, components, events, extra)
+			return
+	# The first two real Volcano rolls have already been made on the common
+	# snapshot by the inherited pair resolver. Add one more disjoint roll.
+	var targets: Array[int] = []
+	var budgets: Array[int] = []
+	for component in components:
+		budgets.append(int(component.get("budget", 0)))
+		for index in component.get("targets", []):
+			if not targets.has(int(index)): targets.append(int(index))
+	var third := _power_component(int(sources[0]["index"]), VOLCANO, targets)
+	budgets.append(int(third.get("budget", 0)))
+	for index in third.get("targets", []):
+		if not targets.has(int(index)): targets.append(int(index))
+	targets.sort()
+	var budget := budgets[0] + budgets[1] + budgets[2]
+	var component := {"power":"mega_volcano", "origin":int(sources[0]["index"]),
+		"kind":VOLCANO, "budget":budget, "targets":targets.duplicate(), "indices":targets.duplicate()}
+	var details := extra.duplicate(true)

+func _mega_arc(source: Vector2, destination: Vector2, flight: float, step: float, ordinal: int) -> Vector2:
