"""D: one normalized vertical-arch endpoint change; no angle randomization."""
from pathlib import Path
import hashlib,importlib.util,json,math
import bpy

ROOT=Path(__file__).resolve().parents[1]
SOURCE=ROOT/'native/INPUT_C.blend';OUTPUT=ROOT/'native/UPRIGHT_TIP_D.blend'
PIN='afd0c35e0450bd46c26a8cd5c65762510ef409b738454c410b5ea614ab0daee5'
RECEIPT_PIN='eaaf52f14c12d2858095f002dac75844f982292374fd3826c91d4d2179184aab'
TIP_ENDPOINT=.90
spec=importlib.util.spec_from_file_location('records',ROOT/'tools/record_helpers.py')
p=importlib.util.module_from_spec(spec);spec.loader.exec_module(p);g=p.g
def sha(data):return hashlib.sha256(data).hexdigest()
def dump(name,data):(ROOT/name).write_bytes((json.dumps(data,indent=2)+'\n').encode())

def frozen(mesh,allowed_z):
    data=p.mesh_prefix(mesh)
    for i in allowed_z:data['vertices'][i][2]=None
    return g.fingerprint(data)

def render(scene,name,detail=False):
    old=(scene.render.filepath,scene.cycles.samples,scene.render.resolution_x,scene.render.resolution_y,
         scene.render.use_border,scene.render.use_crop_to_border,scene.render.border_min_x,
         scene.render.border_max_x,scene.render.border_min_y,scene.render.border_max_y)
    scene.cycles.samples=128
    if detail:
        scene.render.resolution_x=3456;scene.render.resolution_y=7680
        scene.render.use_border=True;scene.render.use_crop_to_border=True
        scene.render.border_min_x=61/432;scene.render.border_max_x=125/432
        scene.render.border_min_y=1-492/960;scene.render.border_max_y=1-428/960
    path=ROOT/'outputs'/name;scene.render.filepath=str(path)
    assert 'FINISHED' in bpy.ops.render.render(write_still=True)
    (scene.render.filepath,scene.cycles.samples,scene.render.resolution_x,scene.render.resolution_y,
     scene.render.use_border,scene.render.use_crop_to_border,scene.render.border_min_x,
     scene.render.border_max_x,scene.render.border_min_y,scene.render.border_max_y)=old
    return {'path':path.relative_to(ROOT).as_posix(),'bytes':path.stat().st_size,
            'sha256':sha(path.read_bytes()),'width':512 if detail else 432,
            'height':512 if detail else 960,'samples':128}

def main():
    assert bpy.app.background and sha(SOURCE.read_bytes())==PIN and not OUTPUT.exists()
    parent_path=ROOT/'documentation/INPUT_C_RECEIPT.json'
    assert sha(parent_path.read_bytes())==RECEIPT_PIN
    prior=json.loads(parent_path.read_text(encoding='utf-8'));assert prior['candidate_sha256']==PIN
    bpy.context.preferences.filepaths.save_version=0
    bpy.ops.wm.open_mainfile(filepath=str(SOURCE));scene=bpy.context.scene;scene.frame_set(1)
    bpy.context.view_layer.update();body=bpy.data.objects['Reference_Grass_Carpet'];mesh=body.data
    assert (len(mesh.vertices),len(mesh.polygons))==(124725,78674)
    names=set(scene.objects.keys());protected={o.name:p.object_record(o) for o in scene.objects if o!=body}
    settings=g.settings_record(scene);materials=g.original_materials()
    identity={k:p.object_record(body)[k] for k in ('matrix_world','matrix_basis','parent','collections','materials')}
    assert prior['parameters']['z_bezier_controls']==[0,.95,1.22,.72]
    times=prior['parameters']['rows_t'];across=prior['parameters']['across']
    rows=sorted(prior['leaves'],key=lambda row:row['blade']);assert len(rows)==73
    allowed=set();bases={};starts={}
    for order,row in enumerate(rows):
        start=row['blade']*16+order*45;starts[row['blade']]=start
        bases[row['blade']]=[tuple(mesh.vertices[start+j].co) for j in range(5)]
        allowed.update(range(start+5,start+61))
    before=frozen(mesh,allowed);old_z={i:mesh.vertices[i].co.z for i in allowed}
    tip_deltas=[]
    for row in rows:
        start=starts[row['blade']];height=row['height_m'];width=row['width_m'];root_z=row['root'][2]
        raw=[]
        for t in times:
            z=3*(1-t)**2*t*.95+3*(1-t)*t*t*1.22+t**3*TIP_ENDPOINT
            raw.extend(height*z+width*.09*math.sin(math.pi*t)*(1-a*a) for a in across)
        raw.append(height*TIP_ENDPOINT);scale=height/max(raw)
        assert len(raw)==61
        for j,z in enumerate(raw):
            if j>=5:mesh.vertices[start+j].co.z=root_z+z*scale
        tip_deltas.append(mesh.vertices[start+60].co.z-old_z[start+60])
    mesh.update()
    def verify():
        bpy.context.view_layer.update()
        assert set(scene.objects.keys())==names
        assert all(p.object_record(bpy.data.objects[name])==record for name,record in protected.items())
        assert g.settings_record(scene)==settings and g.original_materials()==materials
        assert {k:p.object_record(body)[k] for k in identity}==identity
        assert frozen(body.data,allowed)==before
        assert int(body['blade_count'])==7590 and int(body['tuft_count'])==2559
        for row in rows:
            start=starts[row['blade']];co=[x.co for x in body.data.vertices[start:start+61]]
            assert [tuple(x) for x in co[:5]]==bases[row['blade']]
            assert abs(max(x.z for x in co)-row['root'][2]-row['height_m'])<1e-6
            assert abs((co[34]-co[30]).length-row['width_m'])<1e-6
            assert min(x.z for x in co)>=row['root'][2]-1e-6
    verify();print('UPRIGHT_FROZEN_RECORDS_PASS',len(protected),flush=True)
    body['probe_shape']='D:vertical arch endpoint .72 to .90;XY/topology/colour fixed'
    note=bpy.data.texts.new('START_HERE_UPRIGHT_TIP_D')
    note.write('Native D versus C. One logical normalized vertical-arch endpoint changes .72→.90. Only selected leaf Z changes;allXY/colours/topology/base vertices/other scene records fixed. No random rotation or density change. C and B notes describe historical inputs. No game integration.\n')
    assert 'FINISHED' in bpy.ops.wm.save_as_mainfile(filepath=str(OUTPUT),check_existing=False)
    output_sha=sha(OUTPUT.read_bytes());bpy.ops.wm.open_mainfile(filepath=str(OUTPUT));scene=bpy.context.scene;body=bpy.data.objects['Reference_Grass_Carpet']
    verify()
    images=[im for im in bpy.data.images if im.source not in ('GENERATED','VIEWER')]
    assert len(images)==5 and all(im.packed_file for im in images) and not bpy.data.libraries
    full=render(scene,'01_D_UPRIGHT_432x960.png');detail=render(scene,'03_D_DETAIL_8X.png',True);verify()
    controls=[]
    for name,w,h,source_name in [('00_C_FULL_432x960.png',432,960,'outputs/01_C_SOFT_432x960.png'),
                                 ('02_C_DETAIL_8X.png',512,512,'outputs/03_C_DETAIL_8X.png')]:
        path=ROOT/'outputs'/name;match=next(x for x in prior['renders'] if x['path']==source_name)
        assert sha(path.read_bytes())==match['sha256'] and path.stat().st_size==match['bytes']
        controls.append({'path':path.relative_to(ROOT).as_posix(),'bytes':path.stat().st_size,
                         'sha256':sha(path.read_bytes()),'width':w,'height':h,'samples':128})
    changed=sum(body.data.vertices[i].co.z!=z for i,z in old_z.items())
    assert changed>0 and min(tip_deltas)>0
    receipt={'status':'NATIVE_UPRIGHT_TIP_SAVED_REOPENED_RENDERED','source_sha256':PIN,
             'candidate_sha256':output_sha,'builder_sha256':sha(Path(__file__).read_bytes()),
             'tool':'Blender5.2.2 LTS','logical_parameter':{'name':'normalized_arch_endpoint','before':.72,'after':TIP_ENDPOINT},
             'selected_leaves':73,'selected_tufts':12,'total_leaves':7590,'total_tufts':2559,
             'vertices':124725,'faces':78674,'triangles':142022,'allowed_z_vertices':len(allowed),'changed_z_vertices':changed,
             'tip_rise_m':{'min':min(tip_deltas),'max':max(tip_deltas)},
             'mesh_frozen_except_declared_z_sha256':before,'all_xy_colours_topology_other_z_unchanged':True,
             'planted_first_five_vertices_per_leaf_exact':True,'protected_non_body_records':len(protected),
             'camera_lighting_material_records_unchanged':True,'max_height_width_tolerance_m':1e-6,
             'packed_images':len(images),'linked_libraries':0,'angles_or_randomness_changed':False,
             'footprint_limit':'XY exact,so inherited C point-footprint test unchanged;no new ray test or triangle/leaf-leaf proof.',
             'hypothesis':'Reduced droop via one normalized Z-arch control;not external rotation/random-tilt proposal.',
             'renders':[controls[0],full,controls[1],detail],'game_integration':'NOT_PERFORMED','godot_tests':'NOT_RUN',
             'human_choice':'PENDING','android_performance':'NOT_MEASURED'}
    assert sha(SOURCE.read_bytes())==PIN and sha(OUTPUT.read_bytes())==output_sha
    dump('documentation/UPRIGHT_TIP_RECEIPT.json',receipt)
    print('UPRIGHT_READY '+json.dumps({k:receipt[k] for k in ['status','changed_z_vertices','tip_rise_m','candidate_sha256']}))

if __name__=='__main__':main()
