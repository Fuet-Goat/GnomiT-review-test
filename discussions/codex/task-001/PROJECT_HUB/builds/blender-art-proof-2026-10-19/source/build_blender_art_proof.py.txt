def mat(name,hexcolor,rough=.7,metal=0,emission=0):
    rgb=tuple(int(hexcolor[i:i+2],16)/255 for i in (0,2,4))
    c=tuple(v/12.92 if v<=.04045 else ((v+.055)/1.055)**2.4 for v in rgb)
    m=bpy.data.materials.new(name);m.diffuse_color=(*c,1);m.use_nodes=True
    p=m.node_tree.nodes.get('Principled BSDF');p.inputs['Base Color'].default_value=(*c,1)
    p.inputs['Roughness'].default_value=rough;p.inputs['Metallic'].default_value=metal
    if emission:
        p.inputs['Emission Color'].default_value=(*c,1);p.inputs['Emission Strength'].default_value=emission
    return m

def finish(obj,name,material,bevel=0,smooth=True):
    obj.name=name
    if material:obj.data.materials.append(material)
    if bevel:
        m=obj.modifiers.new('Rounded material edges','BEVEL');m.width=bevel;m.segments=3
        m=obj.modifiers.new('Weighted highlights','WEIGHTED_NORMAL')
    if smooth and obj.type=='MESH':
        for face in obj.data.polygons:face.use_smooth=True
    return obj

def cube(name,loc,scale,material,bevel=.06):
    verts=[(x*scale[0]*.5,y*scale[1]*.5,z*scale[2]*.5) for z in [-1,1] for y in [-1,1] for x in [-1,1]]
    faces=[(0,2,3,1),(4,5,7,6),(0,1,5,4),(2,6,7,3),(0,4,6,2),(1,3,7,5)]
    me=bpy.data.meshes.new(name);me.from_pydata(verts,[],faces);me.update()
    o=bpy.data.objects.new(name,me);bpy.context.collection.objects.link(o);o.location=loc
    return finish(o,name,material,bevel,False)

sphere_cache={}
def ball(name,loc,scale,material,sub=2):
    if sub not in sphere_cache:
        bm=bmesh.new();bmesh.ops.create_icosphere(bm,subdivisions=sub,radius=1)
        me=bpy.data.meshes.new('Rounded stone/foliage base');bm.to_mesh(me);bm.free();sphere_cache[sub]=me
    o=bpy.data.objects.new(name,sphere_cache[sub].copy());bpy.context.collection.objects.link(o);o.location=loc;o.scale=scale
    return finish(o,name,material,0,True)

def tube(name,points,radius,material):
    c=bpy.data.curves.new(name,'CURVE');c.dimensions='3D';c.resolution_u=16;c.bevel_depth=radius;c.bevel_resolution=3
    sp=c.splines.new('BEZIER');sp.bezier_points.add(len(points)-1)
    for b,p in zip(sp.bezier_points,points):b.co=p;b.handle_left_type='AUTO';b.handle_right_type='AUTO'
    o=bpy.data.objects.new(name,c);bpy.context.collection.objects.link(o);o.data.materials.append(material);return o

def leaf(name,start,end,width,material):
    a=Vector(start);b=Vector(end);d=b-a;n=Vector((-d.y,d.x,0)).normalized()*width
    mid=a+d*.48+Vector((0,0,width*.55))
    verts=[a,mid+n,b,mid-n,mid+Vector((0,0,.045))]
    faces=[(0,1,4),(1,2,4),(2,3,4),(3,0,4)]
    mesh=bpy.data.meshes.new(name);mesh.from_pydata(verts,[],faces);mesh.update()
    o=bpy.data.objects.new(name,mesh);bpy.context.collection.objects.link(o);finish(o,name,material)
    solid=o.modifiers.new('Leaf thickness','SOLIDIFY');solid.thickness=.012
    sub=o.modifiers.new('Rounded leaf blade','SUBSURF');sub.levels=1;return o
