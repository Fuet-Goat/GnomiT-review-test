"""R2 sleeve-only skinning repair, keeping the accepted motion and form.

Uses the existing coat's surface graph. No object/bone/action/material or
modifier changes, and no added groups. Canonical front is -Y, right is -X.
"""
import struct
import hashlib

import numpy as np


GROUPS = ("Root", "Chest", "upperarm.R", "forearm.R")


def _smooth(a, b, value):
    t = max(0., min(1., (value - a) / (b - a)))
    return t * t * (3. - 2. * t)


def _weight_snapshot(coat):
    return [tuple((int(g.group), float(g.weight)) for g in v.groups)
            for v in coat.data.vertices]


def _snapshot_hash(snapshot, indices):
    h = hashlib.sha256()
    for i in indices:
        h.update(struct.pack('<II', i, len(snapshot[i])))
        for group, weight in snapshot[i]:
            h.update(struct.pack('<If', group, weight))
    return h.hexdigest()


def apply_sleeve_weights(coat, rig=None):
    """Repair only the right sleeve field; return JSON-compatible evidence."""
    assert coat.name == "R2_ContinuousCoat" and coat.type == "MESH"
    assert [g.name for g in coat.vertex_groups] == [
        'Root', 'Chest', 'upperarm.R', 'forearm.R', 'upperarm.L', 'forearm.L']
    modifiers = [m for m in coat.modifiers if m.type == 'ARMATURE']
    assert len(modifiers) == 1 and modifiers[0].use_deform_preserve_volume
    if rig is not None:
        assert modifiers[0].object == rig
    original = _weight_snapshot(coat)
    n = len(original)
    coordinates = np.array([tuple(v.co) for v in coat.data.vertices], dtype=np.float64)
    to_root = coat.parent.matrix_world.inverted() @ coat.matrix_world
    # The pinned coat is already canonical, but respect its unchanged local
    # transform when selecting the right sleeve field.
    coordinates = np.array([tuple(to_root @ v.co) for v in coat.data.vertices], dtype=np.float64)
    x, y, z = coordinates.T
    old = np.zeros((n, 4), dtype=np.float64)
    group_ids = {coat.vertex_groups[name].index: k for k, name in enumerate(GROUPS)}
    for i, memberships in enumerate(original):
        for group, weight in memberships:
            if group in group_ids:
                old[i, group_ids[group]] = weight

    # Only the right upper garment and its shoulder/underarm transition.
    # All centre/left, lower hem and the non-sleeve low torso are Dirichlet
    # boundaries retaining their original Root/Chest breathing blend exactly.
    repair = (x < -.03) & (z > .54)
    torso_anchor = (x > -.14) & (z < .83)
    writable = repair & ~torso_anchor
    wrist = np.array((-.350, -.100, .680))
    wrist_distance = np.linalg.norm(coordinates - wrist, axis=1)
    # The entire accepted cuff is forearm-bound, including its underside.
    cuff = writable & (wrist_distance < .105) & (x < -.28)
    # The outer shoulder is an upper-arm anchor. The intervening surface
    # solves continuously rather than branching at nearest-segment bisectors.
    shoulder = writable & (x < -.285) & (z > .835) & ~cuff
    # R1's old oblique mask also bound the neck's inner shoulder to the arm.
    # This is the actual .04x compression/overlap site. It belongs to Chest;
    # the surface solve carries that anchor smoothly toward the outer sleeve.
    neck = writable & (x > -.145) & (z > .955)
    fixed = ~writable | cuff | shoulder | neck
    values = old.copy()
    values[cuff] = (0., 0., 0., 1.)
    values[shoulder] = (0., 0., 1., 0.)
    values[neck] = (0., 1., 0., 0.)

    edges = np.array([tuple(e.vertices) for e in coat.data.edges], dtype=np.int32)
    starts = np.concatenate((edges[:, 0], edges[:, 1]))
    ends = np.concatenate((edges[:, 1], edges[:, 0]))
    lengths = np.linalg.norm(coordinates[starts] - coordinates[ends], axis=1)
    conductance = 1. / np.maximum(lengths, .001)
    degree = np.bincount(starts, weights=conductance, minlength=n)
    assert np.all(degree > 0)
    free = np.flatnonzero(~fixed)
    residual = float('inf')
    iterations = 0
    for iteration in range(3400):
        next_values = np.column_stack([
            np.bincount(starts, weights=conductance * values[ends, k], minlength=n) / degree
            for k in range(4)])
        residual = float(np.max(np.abs(next_values[free] - values[free])))
        values[free] = next_values[free]
        iterations = iteration + 1
        if residual < 2e-7:
            break
    # Hard constraints remain exact; normalise only vertices in this field.
    targets = np.flatnonzero(writable)
    values[targets] = np.maximum(values[targets], 0.)
    values[targets] /= values[targets].sum(axis=1)[:, None]
    groups = [coat.vertex_groups[name] for name in GROUPS]
    changed = []
    for i in targets.tolist():
        desired = [float(w) for w in values[i]]
        if max(abs(desired[k] - old[i, k]) for k in range(4)) > 2e-7:
            for group in groups:
                group.remove([i])
            for group, weight in zip(groups, desired):
                if weight > 1e-8:
                    group.add([i], weight, 'REPLACE')
            changed.append(i)
    final = _weight_snapshot(coat)
    left_indices = np.flatnonzero(x >= 0).tolist()
    outside = np.flatnonzero(~writable).tolist()
    assert _snapshot_hash(original, left_indices) == _snapshot_hash(final, left_indices)
    assert _snapshot_hash(original, outside) == _snapshot_hash(final, outside)
    assert changed
    maximum_error = max(abs(sum(w for _g, w in final[i]) - 1.) for i in changed)
    assert maximum_error < 2e-6
    pts = coordinates[changed]
    return {
        'method': 'inverse-edge-length surface-graph harmonic Dirichlet weights',
        'group_names': list(GROUPS), 'new_groups': 0, 'helper_bones': 0,
        'preserve_volume': True, 'changed_vertex_count': len(changed),
        'changed_canonical_bounds': {'min': pts.min(axis=0).tolist(), 'max': pts.max(axis=0).tolist()},
        'left_unchanged_vertex_count': len(left_indices),
        'left_weights_sha256_before': _snapshot_hash(original, left_indices),
        'left_weights_sha256_after': _snapshot_hash(final, left_indices),
        'outside_field_unchanged_vertex_count': len(outside),
        'outside_field_weights_sha256_before': _snapshot_hash(original, outside),
        'outside_field_weights_sha256_after': _snapshot_hash(final, outside),
        'maximum_weight_sum_error': maximum_error,
        'harmonic_iterations': iterations, 'harmonic_maximum_step_residual': residual,
        'forearm_cuff_anchors': int(np.count_nonzero(cuff)),
        'upperarm_shoulder_anchors': int(np.count_nonzero(shoulder)),
        'chest_inner_shoulder_anchors': int(np.count_nonzero(neck)),
        'free_surface_vertices': len(free),
    }
