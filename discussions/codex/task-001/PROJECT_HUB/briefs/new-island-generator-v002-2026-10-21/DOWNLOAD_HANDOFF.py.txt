"""Download/verify v002 generator handoff/reference-image package data into a NEW folder; never runs project code.
python DOWNLOAD_HANDOFF.py --output NEW_FOLDER
"""
import argparse,base64,hashlib,io,json,stat,urllib.request,zipfile
from pathlib import Path,PurePosixPath
REPO='Fuet-Goat/GnomiT-review-test'
PREFIX='discussions/codex/task-001/PROJECT_HUB/briefs/new-island-generator-v002-2026-10-21/'
def get(url):
 req=urllib.request.Request(url,headers={'User-Agent':'GnomiT-readonly-art-review'})
 with urllib.request.urlopen(req,timeout=90) as r:return r.read()
def main():
 parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('--output',type=Path,required=True);args=parser.parse_args()
 dest=args.output.resolve()
 if dest.exists():raise ValueError('Choose a new evidence directory')
 head=json.loads(get('https://api.github.com/repos/'+REPO+'/git/ref/heads/codex/codex/task-001'))['object']['sha']
 raw='https://raw.githubusercontent.com/'+REPO+'/'+head+'/'+PREFIX
 receipt=json.loads(get(raw+'PACKAGE_RECEIPT.json.txt'))
 payload=base64.b64decode(get(raw+'BRIEF_WITH_REFERENCES.base64.txt'))
 if len(payload)!=receipt['bytes'] or hashlib.sha256(payload).hexdigest()!=receipt['sha256']:raise ValueError('Archive identity mismatch')
 z=zipfile.ZipFile(io.BytesIO(payload))
 if z.testzip() is not None:raise ValueError('ZIP CRC failure')
 manifest=json.loads(z.read('MANIFEST.json'))
 if len(z.namelist())!=len(manifest['files'])+1 or len(z.namelist())!=receipt['entries']:raise ValueError('Unexpected entries')
 for item in z.infolist():
  name=PurePosixPath(item.filename)
  if name.is_absolute() or '..' in name.parts or chr(92) in item.filename or ':' in item.filename or stat.S_ISLNK(item.external_attr>>16):raise ValueError('Unsafe ZIP path')
 for row in manifest['files']:
  data=z.read(row['path'])
  if len(data)!=row['bytes'] or hashlib.sha256(data).hexdigest()!=row['sha256']:raise ValueError('Payload mismatch:'+row['path'])
 dest.mkdir(parents=True,exist_ok=False);z.extractall(dest)
 proof={'status':'PASS','source_commit':head,'zip_sha256':receipt['sha256'],'entries':len(z.namelist()),'no_engine_or_project_code_executed':True}
 (dest/'DOWNLOAD_VERIFIED.json').write_bytes((json.dumps(proof,indent=2)+'\n').encode());print(json.dumps(proof,indent=2))
if __name__=='__main__':main()
