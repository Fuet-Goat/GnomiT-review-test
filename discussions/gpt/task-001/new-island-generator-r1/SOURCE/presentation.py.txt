from __future__ import annotations

def scene_points(scene):
    bpy.context.view_layer.update()
    depsgraph=bpy.context.evaluated_depsgraph_get()
    points=[]
    for obj in scene.objects:
        if obj.type!='MESH' or obj.hide_render: continue
        evaluated=obj.evaluated_get(depsgraph)
        points.extend(evaluated.matrix_world @ Vector(corner) for corner in evaluated.bound_box)
    if not points: raise RuntimeError('No renderable native geometry was constructed')
    return points


def configure(scene,cols,cfg):
    scene.unit_settings.system='METRIC'; scene.unit_settings.scale_length=1.0
    scene.render.engine='CYCLES'
    scene.cycles.device=cfg['render']['device']
    scene.cycles.use_denoising=cfg['render']['denoise']
    scene.render.resolution_x=432;scene.render.resolution_y=960
    scene.render.resolution_percentage=100
    scene.render.pixel_aspect_x=1.0;scene.render.pixel_aspect_y=1.0
    scene.render.image_settings.file_format='PNG'
    scene.render.image_settings.color_mode='RGBA'
    scene.render.image_settings.color_depth='8'
    scene.render.film_transparent=cfg['render']['transparent']
    scene.view_settings.view_transform=cfg['render']['view_transform']
    scene.view_settings.look=cfg['render']['look']
    scene.view_settings.exposure=cfg['render']['exposure']
    scene.view_settings.gamma=cfg['render']['gamma']
    scene.render.fps=24
    scene.frame_start=1;scene.frame_end=1
    scene.world=bpy.data.worlds.new('Airy_Soft_Sky')
    scene.world.use_nodes=True
    bg=scene.world.node_tree.nodes.get('Background')
    bg.inputs['Color'].default_value=(*rgb(cfg['palette']['sky']),1.0)
    bg.inputs['Strength'].default_value=0.48
    lights=[('Key_Warm',(-5.5,-4.5,12.0),1250.0,7.0,(1.0,0.88,0.72)),
            ('Fill_Sky',(6,-1,8.5),480.0,8.0,(0.80,0.89,1.0)),
            ('Rim_Soft',(-2,9,11),780.0,7.0,(1.0,0.95,0.83))]
    for name,pos,energy,size,color in lights:
        data=bpy.data.lights.new(name,'AREA');data.energy=energy;data.shape='DISK';data.size=size;data.color=color
        obj=bpy.data.objects.new(name,data);cols['LIGHTS'].objects.link(obj);obj.location=pos
        obj.rotation_euler=(Vector((0,0,0.6))-obj.location).to_track_quat('-Z','Y').to_euler()


def fit_camera(scene,cols,settings):
    points=scene_points(scene)
    elev=math.radians(settings['elevation_degrees']); az=math.radians(settings['azimuth_degrees'])
    outward=Vector((math.sin(az)*math.cos(elev),-math.cos(az)*math.cos(elev),math.sin(elev)))
    rotation=(-outward).to_track_quat('-Z','Y')
    right=rotation @ Vector((1,0,0)); up=rotation @ Vector((0,1,0))
    xs=[p.dot(right) for p in points]; ys=[p.dot(up) for p in points]; ds=[p.dot(outward) for p in points]
    centre=right*((min(xs)+max(xs))/2)+up*((min(ys)+max(ys))/2)+outward*((min(ds)+max(ds))/2)
    data=bpy.data.cameras.new('Portrait_Camera');data.type='ORTHO';data.sensor_fit='VERTICAL'
    ratio=432/960; margin=settings['margin_fraction']
    if not 0.01<=margin<0.2:raise ValueError('Invalid camera margin')
    scale=max(max(ys)-min(ys),(max(xs)-min(xs))/ratio)/(1-2*margin)
    data.ortho_scale=scale;data.clip_start=0.05;data.clip_end=200
    cam=bpy.data.objects.new('Portrait_Camera',data);cols['CAMERAS'].objects.link(cam)
    cam.location=centre+outward*settings['distance'];cam.rotation_euler=rotation.to_euler()
    scene.camera=cam
    # Vertical sensor fit is explicit; this is a mathematical framing check.
    uv=[(0.5+(p-centre).dot(right)/(scale*ratio),0.5+(p-centre).dot(up)/scale) for p in points]
    bbox=[min(v[0] for v in uv),min(v[1] for v in uv),max(v[0] for v in uv),max(v[1] for v in uv)]
    if min(bbox[:2])<0 or max(bbox[2:])>1:raise RuntimeError('Camera fitting failed')
    return {'name':cam.name,'type':'ORTHO','sensor_fit':'VERTICAL','location':list(cam.location),
            'rotation_euler_radians':list(cam.rotation_euler),'ortho_scale':scale,
            'projected_geometry_bounds_normalized':bbox,'margin_input':margin,
            'framing_check':'Mathematical bounds only; actual rendered framing still needs observation'}


def geometry_metadata(scene):
    points=scene_points(scene)
    counts={}
    for obj in scene.objects: counts[obj.type]=counts.get(obj.type,0)+1
    meshes=[o for o in scene.objects if o.type=='MESH']
    for obj in meshes:
        for v in obj.data.vertices:
            if not all(math.isfinite(x) for x in v.co):
                raise RuntimeError('Nonfinite vertex in '+obj.name)
        if any(len(p.vertices)<3 or p.area<1e-12 for p in obj.data.polygons):
            raise RuntimeError('Degenerate polygon in '+obj.name)
    return {'objects_by_type':counts,'mesh_object_count':len(meshes),
            'base_mesh_vertices':sum(len(o.data.vertices) for o in meshes),
            'base_mesh_polygons':sum(len(o.data.polygons) for o in meshes),
            'base_mesh_estimated_triangles':sum(len(p.vertices)-2 for o in meshes for p in o.data.polygons),
            'evaluated_bounds_min':[min(p[a] for p in points) for a in range(3)],
            'evaluated_bounds_max':[max(p[a] for p in points) for a in range(3)],
            'topology_limit':'Counts before modifier triangulation are not exported triangle or mobile performance measurements'}


def export_glb(scene,filepath):
    for obj in scene.objects:
        obj.select_set(False)
    for obj in scene.objects:
        if obj.type in {'MESH','EMPTY','CAMERA'} and not obj.get('exclude_from_glb',False):
            obj.select_set(True)
    operator=bpy.ops.export_scene.gltf
    required={'filepath','export_format','use_selection','export_apply',
              'export_cameras','export_lights','export_animations','export_extras'}
    available={p.identifier for p in operator.get_rna_type().properties}
    missing=required-available
    if missing: raise RuntimeError('glTF exporter API missing: '+','.join(sorted(missing)))
    result=operator(filepath=str(filepath),export_format='GLB',use_selection=True,
                    export_apply=True,export_cameras=True,export_lights=False,
                    export_animations=False,export_extras=True)
    if 'FINISHED' not in result:raise RuntimeError('GLB export did not finish')
    payload=filepath.read_bytes()
    if len(payload)<20:raise RuntimeError('Truncated GLB')
    magic,version,length=struct.unpack_from('<4sII',payload,0)
    if magic!=b'glTF' or version!=2 or length!=len(payload):raise RuntimeError('Invalid GLB header')
    offset=12; document=None
    while offset<len(payload):
        if offset+8>len(payload):raise RuntimeError('Truncated GLB chunk header')
        n,kind=struct.unpack_from('<II',payload,offset);offset+=8
        if offset+n>len(payload):raise RuntimeError('GLB chunk out of bounds')
        if kind==0x4E4F534A:document=json.loads(payload[offset:offset+n].decode('utf-8').rstrip(' \x00'))
        offset+=n
    if document is None:raise RuntimeError('GLB JSON chunk not found')
    names={n.get('name') for n in document.get('nodes',[])}
    required_names={'Road_Centerline','Level_1_Anchor','Level_2_Anchor','Portrait_Camera'}
    if not required_names<=names:raise RuntimeError('Missing GLB route/anchor/camera nodes: '+str(required_names-names))
    for record in document.get('images',[])+document.get('buffers',[]):
        if 'uri' in record and not record['uri'].startswith('data:'):
            raise RuntimeError('GLB references an external resource')
    return {'header_valid':True,'node_count':len(document.get('nodes',[])),
            'named_anchors_present':True,'external_uri_count':0,
            'reimport':'NOT_RUN','lighting_note':'Area lights/world/Cycles look excluded; recreate lighting after engine import'}


def render(scene,path,width,height,samples):
    scene.render.resolution_x=width;scene.render.resolution_y=height
    scene.cycles.samples=int(samples);scene.render.filepath=str(path)
    result=bpy.ops.render.render(write_still=True)
    if 'FINISHED' not in result:raise RuntimeError('Render operator did not finish')
    data=path.read_bytes()
    if data[:8]!=b'\x89PNG\r\n\x1a\n' or len(data)<24:
        raise RuntimeError('PNG output missing or truncated')
    actual=struct.unpack('>II',data[16:24])
    if actual!=(width,height):raise RuntimeError('Wrong render dimensions')
    return {'path':path.name,'width':width,'height':height,'samples':samples,
            'operator_finished':True,'png_header_verified':True,'visual_acceptance':'PENDING_HUMAN_REVIEW'}
